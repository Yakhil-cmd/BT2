### Title
Unbounded Private-Payment Chain Length Enables CPU/Memory Exhaustion DoS - (File: wallet.js, indivisible_asset.js, private_payment.js)

### Summary
`handlePrivatePaymentChains()` in `wallet.js` accepts `body.chains` from a private-payment counterparty and only checks that each chain is a non-empty array of well-formed objects, without any bound on chain length (`arrPrivateElements.length`) or number of chains (`arrChains.length`) [1](#0-0) . Each chain is then walked element-by-element in `parsePrivatePaymentChain()` in `indivisible_asset.js`, performing a DB query and a SHA-256/ECDSA spend-proof hash computation per element with no cap on the number of elements [2](#0-1) .

### Finding Description
This is analogous to CVE-2017-14055: FFmpeg's `mv_read_header()` trusted an attacker-supplied `nb_frames` count and looped over it without checking that backing data actually existed or was bounded, letting CPU/memory blow up. In ocore, `handlePrivatePaymentChains()` validates only structural shape of each chain element (`isNonemptyObject`, presence of `unit`, `payload.asset`, `inputs`, `outputs`), but never checks `arrChains.length` or any individual `arrPrivateElements.length` against a sane maximum [1](#0-0) . Each chain element goes through `network.handleOnlinePrivatePayment()` → `privatePayment.validateAndSavePrivatePaymentChain()` → `indivisibleAsset.validateAndSavePrivatePaymentChain()` → `parsePrivatePaymentChain()`, which does `async.forEachOfSeries` over the entire supplied `arrPrivateElements` array, running `validatePrivatePayment()` for every element [3](#0-2) . `validatePrivatePayment()` computes a spend-proof hash via `objectHash.getBase64Hash` and (through `validation.validatePayment`) runs full definition/signature evaluation for the associated address for every one of these attacker-supplied elements [4](#0-3) . Because the number of elements per chain (and number of chains per message) is attacker-controlled and unbounded, a private-payment counterparty (someone the victim wallet is paired with, exactly the actor class this scan is scoped to) can submit a single `private_payment` / chains message with an extremely long fabricated element array, forcing the receiving node/wallet to perform an unbounded number of DB round-trips and cryptographic hash operations synchronously in series before any length-based rejection occurs.

### Impact Explanation
A malicious private-payment counterparty (or paired device) can force a victim's ocore node/wallet into large CPU and memory consumption while attempting to validate a chain of essentially arbitrary length, since none of `handlePrivatePaymentChains`, `handleOnlinePrivatePayment`, `validateAndSavePrivatePaymentChain`, or `parsePrivatePaymentChain` cap chain length before doing per-element DB/crypto work. This can degrade or stall wallet/hub responsiveness and, if triggered repeatedly, be used to keep a node busy validating bogus chains, matching the "network unable to confirm new units / node resource exhaustion" impact class permitted by the scan rules.

### Likelihood Explanation
Likelihood is moderate-to-high: `handlePrivatePaymentChains` is reachable directly from `wallet.js`'s hub message handler by any paired correspondent device or via the hub relaying arbitrary chain content, and the only pre-checks are shape checks — no length/size cap exists anywhere along the call path before the per-element loop begins.

### Recommendation
Add an explicit upper bound (e.g., a `MAX_PRIVATE_CHAIN_LENGTH` and `MAX_PRIVATE_CHAINS_PER_MESSAGE` constant) and reject in `handlePrivatePaymentChains` (wallet.js) before any per-element processing, mirroring existing caps such as `MAX_MESSAGES_PER_UNIT`/`MAX_INPUTS_PER_PAYMENT_MESSAGE` used elsewhere in the codebase. Also add the same guard as an early check inside `parsePrivatePaymentChain`/`validateAndSavePrivatePaymentChain` (indivisible_asset.js and divisible_asset.js) so that any other code path constructing `arrPrivateElements` from network input is protected, not just the wallet.js entry point.

### Proof of Concept
A paired device (private-payment counterparty) sends, over the existing `private_payment_chains` hub/direct message channel, a `body.chains` payload containing one array with tens of thousands of synthetically well-formed but ultimately invalid private-element objects (satisfying only the shape checks in `wallet.js:959-971`). The receiving wallet enters `parsePrivatePaymentChain`'s `async.forEachOfSeries` and performs one DB query plus hash/signature computation per element before the first validation failure is reported, consuming disproportionate CPU/DB time relative to the tiny message size — the same "declared length vs. actual bound" DoS pattern as CVE-2017-14055's `nb_frames` loop.

function handlePrivatePaymentChains(ws, body, from_address, callbacks){
	var arrChains = body.chains;
	if (!ValidationUtils.isNonemptyArray(arrChains))
		return callbacks.ifError("no chains found");
	if (!arrChains.every(c =>
		isNonemptyArray(c) &&
		c.every(e =>
			isNonemptyObject(e) &&
			isNonemptyString(e.unit) &&
			isNonemptyObject(e.payload) &&
			isNonemptyString(e.payload.asset) &&
			isNonemptyArray(e.payload.inputs) &&
			isNonemptyArray(e.payload.outputs) &&
			e.payload.inputs.every(isNonemptyObject) &&
			e.payload.outputs.every(isNonemptyObject)
		)
	))
		return callbacks.ifError("malformed private chain");

// arrPrivateElements is ordered in reverse chronological order
function parsePrivatePaymentChain(conn, arrPrivateElements, callbacks){
	var bAllStable = true;
	var issuePrivateElement = arrPrivateElements[arrPrivateElements.length-1];
	if (!issuePrivateElement.payload || !issuePrivateElement.payload.inputs || !issuePrivateElement.payload.inputs[0])
		return callbacks.ifError("invalid issue private element");
	var asset = issuePrivateElement.payload.asset;
	if (!asset)
		return callbacks.ifError("no asset in issue private element");
	var denomination = issuePrivateElement.payload.denomination;
	if (!denomination)
		return callbacks.ifError("no denomination in issue private element");
	async.forEachOfSeries(
		arrPrivateElements,
		function(objPrivateElement, i, cb){
			if (!objPrivateElement.payload || !objPrivateElement.payload.inputs || !objPrivateElement.payload.inputs[0])
				return cb("invalid payload");
			if (!objPrivateElement.output)
				return cb("no output in private element");
			if (objPrivateElement.payload.asset !== asset)
				return cb("private element has a different asset");
			if (objPrivateElement.payload.denomination !== denomination)
				return cb("private element has a different denomination");
			var prevElement = null; 
			if (i+1 < arrPrivateElements.length){ // excluding issue transaction
				var prevElement = arrPrivateElements[i+1];
				if (prevElement.unit !== objPrivateElement.payload.inputs[0].unit)
					return cb("not referencing previous element unit");
				if (prevElement.message_index !== objPrivateElement.payload.inputs[0].message_index)
					return cb("not referencing previous element message index");
				if (prevElement.output_index !== objPrivateElement.payload.inputs[0].output_index)
					return cb("not referencing previous element output index");
			}
			validatePrivatePayment(conn, objPrivateElement, prevElement, {
				ifError: cb,
				ifOk: function(bStable, input_address){
					objPrivateElement.bStable = bStable;
					objPrivateElement.input_address = input_address;
					if (!bStable)
						bAllStable = false;
					cb();
				}
			});
		},
		function(err){
			if (err)
				return callbacks.ifError(err);
			callbacks.ifOk(bAllStable);
		}
	);
}
