The reported bug is specific to an ERC-4626-style vault (`ElytraDepositPoolV1`) where a share token (`elyAsset`) price is computed as `totalTVL / totalSupply`, and burning shares before removing the corresponding backing assets from the TVL calculation causes price inflation. NEAR's protocol-level staking/unstaking mechanism has no analogous share/price accounting: an account's `locked` and `amount` fields are updated directly and atomically in `action_stake` [1](#0-0) , and there is no derived "share price" computed from a TVL-style aggregate divided by an outstanding token supply. Native token supply changes are verified directly against burnt/minted balances per block via `verify_total_supply_checked` [2](#0-1) , with no separate pending-withdrawal pool whose assets remain double-counted after a burn, unlike the `_getUnstakingVaultBalance` pattern described in the report. Since nearcore has no vault-style share/asset pricing mechanism reachable from an unprivileged transaction, there is no valid analog for this bug class. [1](#0-0) [2](#0-1)

pub(crate) fn action_stake(
    account: &mut Account,
    result: &mut ActionResult,
    account_id: &AccountId,
    stake: &StakeAction,
    last_block_hash: &CryptoHash,
    epoch_info_provider: &dyn EpochInfoProvider,
) -> Result<(), RuntimeError> {
    let increment = stake.stake.saturating_sub(account.locked());

    if let Some(new_balance) = account.amount().checked_sub(increment) {
        if account.locked().is_zero() && stake.stake.is_zero() {
            // if the account hasn't staked, it cannot unstake
            result.result =
                Err(ActionErrorKind::TriesToUnstake { account_id: account_id.clone() }.into());
            return Ok(());
        }

        if stake.stake > Balance::ZERO {
            let minimum_stake = epoch_info_provider.minimum_stake(last_block_hash)?;
            if stake.stake < minimum_stake {
                result.result = Err(ActionErrorKind::InsufficientStake {
                    account_id: account_id.clone(),
                    stake: stake.stake,
                    minimum_stake,
                }
                .into());
                return Ok(());
            }
        }

        result.validator_proposals.push(ValidatorStake::new(
            account_id.clone(),
            stake.public_key.clone(),
            stake.stake,
        ));
        if stake.stake > account.locked() {
            // We've checked above `account.amount >= increment`
            account.set_amount(new_balance);
            account.set_locked(stake.stake);
        }
    } else {
        result.result = Err(ActionErrorKind::TriesToStake {
            account_id: account_id.clone(),
            stake: stake.stake,
            locked: account.locked(),
            balance: account.amount(),
        }
        .into());
    }
    Ok(())

    pub fn verify_total_supply_checked(
        &self,
        prev_total_supply: Balance,
        minted_amount: Option<Balance>,
    ) -> Option<bool> {
        let mut balance_burnt = Balance::ZERO;

        for chunk in self.chunks().iter_new() {
            balance_burnt = balance_burnt.checked_add(chunk.prev_balance_burnt())?;
        }

        let Some(new_total_supply) = prev_total_supply
            .checked_add(minted_amount.unwrap_or(Balance::ZERO))?
            .checked_sub(balance_burnt)
        else {
            // This corresponds to balance_burnt > prev_total_supply + minted_amount
            // which indicates invalid balance burnt, not arithmetic overflow
            return Some(false);
        };
        Some(self.header().total_supply() == new_total_supply)
    }
