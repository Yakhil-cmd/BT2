### Title
Unbounded recursive hashing of attacker-supplied unit before deep-nesting check causes CPU exhaustion - (File: validation.js, string_utils.js)

### Summary
`validate()` in `validation.js` computes `objectHash.getUnitHash(objUnit)` (line 133) — which recursively walks the *entire* attacker-supplied unit via `getSourceString`/`getJsonSourceString` in `string_utils.js` — **before** the `isTooDeeplyNestedOrHasTooManyNodes(objUnit)` guard is ever evaluated (line 154). Any unprivileged peer that posts a unit (joint) can therefore force full recursive traversal/stringification of an arbitrarily deep or arbitrarily bushy JSON structure embedded in the unit (e.g. inside a `data` message payload, `data_feed` payload, or nested arrays/objects) before any size/depth limit is applied, consuming CPU on every node receiving and validating the joint.

### Finding Description
`validation.js` `validate(objJoint, callbacks, external_conn)`: [1](#0-0) 
computes the unit hash unconditionally on the raw, attacker-controlled `objUnit`, and only afterwards runs the anti-DoS guard: [2](#0-1) 

The hash routines that get invoked recurse into every array/object node without any depth or node-count limit: [3](#0-2) [4](#0-3) 

`getUnitHash` → `getStrippedUnit`/`getNakedUnit` → `getBase64Hash` → `getSourceString`/`getJsonSourceString`, all of which walk the *whole* unit object (including message payloads) with unbounded recursion: [5](#0-4) [6](#0-5) 

`isTooDeeplyNestedOrHasTooManyNodes` (the actual protection) is defined right below `getJsonSourceString` in the same file but is not consulted until after hashing: [7](#0-6) 

An unprivileged unit poster can embed a deeply nested or node-heavy JSON structure in any payload field that is not size-constrained ahead of hashing — e.g. a `data` message (`isNonemptyObject(payload)` is the only check in `validateAADefinition`) or a `data_feed` value — and every node that receives the joint (via `network.js` → `validation.js`) will pay the full recursive-traversal cost of `getSourceString`/`getJsonSourceString` before the cheap `isTooDeeplyNestedOrHasTooManyNodes` short-circuit can reject it. Because the check ordering puts the expensive, unbounded operation first and the guard second, the guard provides no protection against this specific cost.

This mirrors the Wazuh CVE-2026-74039 pattern: a JSON body that is recursively processed before any nesting/size validation, letting an authenticated-but-unprivileged caller (here: any peer able to broadcast a unit) repeatedly trigger CPU-heavy recursive processing.

### Impact Explanation
Every full/light node that receives and validates the joint performs the unbounded recursive hash computation before rejecting the unit, so a single attacker can repeatedly submit crafted units (or joints relayed peer-to-peer) that are cheap to construct but expensive to hash, driving CPU consumption on all nodes that validate the joint and degrading network-wide unit confirmation throughput — a network-availability impact consistent with the "network unable to confirm new units" acceptance criterion.

### Likelihood Explanation
Likelihood is limited by the fact that unit/message payloads are ultimately bounded by other structural checks elsewhere in the validation pipeline (e.g., `data_feed` value/name length caps, `MAX_DENOMINATIONS_PER_ASSET_DEFINITION`, and network message-size limits), so the achievable "nesting depth × node count" before `isTooDeeplyNestedOrHasTooManyNodes` finally runs is constrained by overall message size rather than being unbounded. This reduces exploitability compared to a purely unbounded recursion, but the ordering defect is real: the expensive operation runs first, defeating the purpose of the depth/node-count guard for any payload sub-object that is not independently size-limited (e.g., generic `data` message payloads, which allow arbitrary nested JSON as long as it is a non-empty object).

### Recommendation
Reorder the checks in `validate()` in `validation.js` so `isTooDeeplyNestedOrHasTooManyNodes(objUnit)` (and `isObjectWellFormed`) run **before** `objectHash.getUnitHash(objUnit)` is invoked, ensuring the cheap structural guard rejects oversized/over-nested units prior to any recursive hashing/stringification work.

### Proof of Concept
1. Craft a unit whose `unit.messages[0].payload` (app `data`) is a deeply nested or extremely bushy JSON object (e.g. thousands of nested arrays, or an object with tens of thousands of keys) while keeping the overall wire-serialized size under the network message-size limit.
2. Set `unit.unit` to any value (the hash will not match, but this doesn't matter — the cost is incurred during the *computation* of `getUnitHash`, not from the comparison).
3. Broadcast the joint to a full node. `validate()` calls `objectHash.getUnitHash(objUnit)` at `validation.js:133`, which recurses through the entire nested payload via `getSourceString`/`getJsonSourceString` before `isTooDeeplyNestedOrHasTooManyNodes` at `validation.js:154` ever runs.
4. Repeat with many such joints to consume CPU across all nodes performing joint validation.

function getSourceString(obj) {
	var arrComponents = [];
	function extractComponents(variable){
		if (variable === null)
			throw Error("null value in "+JSON.stringify(obj));
		switch (typeof variable){
			case "string":
				if (variable.includes(STRING_JOIN_CHAR))
					throw Error("00 byte in string value in " + JSON.stringify(obj));
				arrComponents.push("s", variable);
				break;
			case "number":
				if (!isFinite(variable))
					throw Error("invalid number: " + variable);
				arrComponents.push("n", variable.toString());
				break;
			case "boolean":
				arrComponents.push("b", variable.toString());
				break;
			case "object":
				if (Array.isArray(variable)){
					if (variable.length === 0)
						throw Error("empty array in "+JSON.stringify(obj));
					arrComponents.push('[');
					for (var i=0; i<variable.length; i++)
						extractComponents(variable[i]);
					arrComponents.push(']');
				}
				else{
					var keys = Object.keys(variable).sort();
					if (keys.length === 0)
						throw Error("empty object in "+JSON.stringify(obj));
					keys.forEach(function(key){
						if (typeof variable[key] === "undefined")
							throw Error("undefined at "+key+" of "+JSON.stringify(obj));
						if (key.includes(STRING_JOIN_CHAR))
							throw Error("00 byte in object key in " + JSON.stringify(obj));
						arrComponents.push(key);
						extractComponents(variable[key]);
					});
				}
				break;
			default:
				throw Error("getSourceString: unknown type="+(typeof variable)+" of "+variable+", object: "+JSON.stringify(obj));
		}
	}

	extractComponents(obj);
	return arrComponents.join(STRING_JOIN_CHAR);
}

function getJsonSourceString(obj, bAllowEmpty) {
	let cache = new WeakMap();  // object to stringified result
	function stringify(variable){
		if (variable === null)
			throw Error("null value in "+JSON.stringify(obj));
		switch (typeof variable){
			case "string":
				return toWellFormedJsonStringify(variable);
			case "number":
				if (!isFinite(variable))
					throw Error("invalid number: " + variable);
			case "boolean":
				return variable.toString();
			case "object":
				// return cached result if already processed
				if (cache.has(variable))
					return cache.get(variable);
				let result;
				if (Array.isArray(variable)){
					if (variable.length === 0 && !bAllowEmpty)
						throw Error("empty array in "+JSON.stringify(obj));
					result = '[' + variable.map(stringify).join(',') + ']';
				}
				else{
					var keys = Object.keys(variable).sort();
					if (keys.length === 0 && !bAllowEmpty)
						throw Error("empty object in "+JSON.stringify(obj));
					result = '{' + keys.map(function(key){ return toWellFormedJsonStringify(key)+':'+stringify(variable[key]) }).join(',') + '}';
				}
				cache.set(variable, result);  // memoize for future references
				return result;
			default:
				throw Error("getJsonSourceString: unknown type="+(typeof variable)+" of "+variable+", object: "+JSON.stringify(obj));
		}
	}

	return stringify(obj);
}

function isTooDeeplyNestedOrHasTooManyNodes(obj, depthLimit = 100, nodesLimit = 10000) {
	let nodeCount = 0;

	function check(variable, depth){
		if (depth > depthLimit || nodeCount > nodesLimit)
			return true;
		if (variable === null || typeof variable !== "object")
			return false;
		if (Array.isArray(variable)) {
			nodeCount += variable.length;
			for (let v of variable)
				if (check(v, depth + 1))
					return true;
		}
		else {
			nodeCount += Object.keys(variable).length;
			for (let key in variable)
				if (check(variable[key], depth + 1))
					return true;
		}
		return false;
	}

	return check(obj, 1);
}
