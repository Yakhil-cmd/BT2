### Title
Unbounded recursive hashing of unposted unit content causes stack-overflow DoS before depth/size guard is applied - (File: `string_utils.js`, `object_hash.js`, `validation.js`)

### Summary
`validation.js`'s `validate()` computes `objectHash.getUnitHash(objUnit)` to verify the unit hash *before* calling `isTooDeeplyNestedOrHasTooManyNodes(objUnit)`. The hash-computation path (`getUnitHash` → `getStrippedUnit`/`getNakedUnit` → `getBase64Hash`/`getSourceString`) recurses unboundedly over the unit's nested arrays/objects with no depth limit, mirroring the `ixmlNode_free()` recursive-free stack overflow pattern from CVE-2021-28302: unbounded recursion driven directly by untrusted, attacker-controlled input structure.

### Finding Description [1](#0-0)  shows `getUnitHash` is invoked and wrapped in try/catch (which only guards against thrown `Error`s, not stack overflows/process crashes) before any structural safety check runs. The size/depth check `isTooDeeplyNestedOrHasTooManyNodes` is only applied afterward at [2](#0-1) .

`getUnitHash` builds a "stripped"/"naked" unit and passes it into `getBase64Hash`, which calls `getSourceString` (or `getJsonSourceString`) depending on version: [3](#0-2) , [4](#0-3) .

`getSourceString`'s inner `extractComponents` function recurses into every array element / object value with no depth counter and no mechanism (like `setImmediate`) to yield the call stack, unlike other recursive validators in this codebase: [5](#0-4) .

By contrast, `getJsonSourceString` (used for version-2 units) at least has memoization via a `WeakMap` but is still plain unbounded recursion (`stringify` calling itself on nested values without any depth guard): [6](#0-5) .

The codebase clearly recognizes deep-nesting DoS as a threat class and has added explicit depth guards elsewhere (e.g. `MAX_DEPTH` in AA definition validation [7](#0-6) , and `depth > 100` checks in formula validation [8](#0-7) , and `isTooDeeplyNestedOrHasTooManyNodes`/`isTooBigObj` in `string_utils.js` [9](#0-8) ). However, the ordering bug in `validate()` means the unbounded-recursion hash computation runs on the raw, unchecked unit *before* the depth-limiting check is ever applied, defeating the intended mitigation for this specific code path.

### Impact Explanation
Any unprivileged peer/light client can post a joint whose `unit` object contains a deeply nested array or object (e.g., nested `messages[].payload` sub-structures, or any nested field reachable from the naked/stripped unit) to trigger a `RangeError: Maximum call stack size exceeded` inside `getSourceString`/`getJsonSourceString`. Because this happens inside `objectHash.getUnitHash(objUnit)` at the very top of unit validation — the first gate every incoming unit passes through — a crafted unit can crash the validating node's process (denial of service) before any of the size/complexity checks that are supposed to reject such units get a chance to run. Depending on the JS engine, an uncaught `RangeError` thrown deep in a foreign call stack during recursion can, in some cases, escape ordinary try/catch handling reliably or leave the process in an unstable state, and even if caught, repeated submission from any external unit poster can be used to repeatedly try to crash / degrade validation throughput network-wide.

### Likelihood Explanation
High. The attack requires nothing more than assembling a JSON structure with deep nesting (e.g., thousands of nested arrays) inside a message payload field that ends up included in the naked/stripped unit, and posting it as an ordinary unit — reachable by any unprivileged unit poster with no special privileges, keys, or trust relationship. There's no complexity/size check performed prior to the vulnerable hash computation.

### Recommendation
Move the `isTooDeeplyNestedOrHasTooManyNodes(objUnit)` (and `isObjectWellFormed`) checks in `validation.js` to run **before** `objectHash.getUnitHash(objUnit)` is called, so malformed/deeply-nested units are rejected prior to any recursive traversal. Additionally, harden `getSourceString`/`getJsonSourceString` in `string_utils.js` themselves with an internal depth/node-count limit (mirroring `isTooDeeplyNestedOrHasTooManyNodes`) so that any other call site that invokes them directly (there are several, e.g. `getDeviceMessageHashToSign`, `getSignedPackageHashToSign`) is protected regardless of call order.

### Proof of Concept
1. Construct a unit joint with `unit.messages[0].payload` (or another field ultimately included in the naked/stripped unit passed to `getSourceString`) containing extremely deep array nesting, e.g. `JSON.parse('['.repeat(100000) + '1' + ']'.repeat(100000))`.
2. Submit this joint to a node via the normal unit-posting/gossip path so it reaches `validation.js`'s `validate()`.
3. `objectHash.getUnitHash(objUnit)` at line 133 is invoked before the `isTooDeeplyNestedOrHasTooManyNodes` check, causing `getSourceString`'s `extractComponents` to recurse to the same depth, exhausting the call stack and crashing/hanging the node process — a stack overflow analogous to `ixmlNode_free()`'s uncontrolled recursion in CVE-2021-28302.

function getSourceString(obj) {
	var arrComponents = [];
	function extractComponents(variable){
		if (variable === null)
			throw Error("null value in "+JSON.stringify(obj));
		switch (typeof variable){
			case "string":
				if (variable.includes(STRING_JOIN_CHAR))
					throw Error("00 byte in string value in " + JSON.stringify(obj));
				arrComponents.push("s", variable);
				break;
			case "number":
				if (!isFinite(variable))
					throw Error("invalid number: " + variable);
				arrComponents.push("n", variable.toString());
				break;
			case "boolean":
				arrComponents.push("b", variable.toString());
				break;
			case "object":
				if (Array.isArray(variable)){
					if (variable.length === 0)
						throw Error("empty array in "+JSON.stringify(obj));
					arrComponents.push('[');
					for (var i=0; i<variable.length; i++)
						extractComponents(variable[i]);
					arrComponents.push(']');
				}
				else{
					var keys = Object.keys(variable).sort();
					if (keys.length === 0)
						throw Error("empty object in "+JSON.stringify(obj));
					keys.forEach(function(key){
						if (typeof variable[key] === "undefined")
							throw Error("undefined at "+key+" of "+JSON.stringify(obj));
						if (key.includes(STRING_JOIN_CHAR))
							throw Error("00 byte in object key in " + JSON.stringify(obj));
						arrComponents.push(key);
						extractComponents(variable[key]);
					});
				}
				break;
			default:
				throw Error("getSourceString: unknown type="+(typeof variable)+" of "+variable+", object: "+JSON.stringify(obj));
		}
	}

	extractComponents(obj);
	return arrComponents.join(STRING_JOIN_CHAR);

function getJsonSourceString(obj, bAllowEmpty) {
	let cache = new WeakMap();  // object to stringified result
	function stringify(variable){
		if (variable === null)
			throw Error("null value in "+JSON.stringify(obj));
		switch (typeof variable){
			case "string":
				return toWellFormedJsonStringify(variable);
			case "number":
				if (!isFinite(variable))
					throw Error("invalid number: " + variable);
			case "boolean":
				return variable.toString();
			case "object":
				// return cached result if already processed
				if (cache.has(variable))
					return cache.get(variable);
				let result;
				if (Array.isArray(variable)){
					if (variable.length === 0 && !bAllowEmpty)
						throw Error("empty array in "+JSON.stringify(obj));
					result = '[' + variable.map(stringify).join(',') + ']';
				}
				else{
					var keys = Object.keys(variable).sort();
					if (keys.length === 0 && !bAllowEmpty)
						throw Error("empty object in "+JSON.stringify(obj));
					result = '{' + keys.map(function(key){ return toWellFormedJsonStringify(key)+':'+stringify(variable[key]) }).join(',') + '}';
				}
				cache.set(variable, result);  // memoize for future references
				return result;
			default:
				throw Error("getJsonSourceString: unknown type="+(typeof variable)+" of "+variable+", object: "+JSON.stringify(obj));
		}
	}

	return stringify(obj);
}

function isTooDeeplyNestedOrHasTooManyNodes(obj, depthLimit = 100, nodesLimit = 10000) {
	let nodeCount = 0;

	function check(variable, depth){
		if (depth > depthLimit || nodeCount > nodesLimit)
			return true;
		if (variable === null || typeof variable !== "object")
			return false;
		if (Array.isArray(variable)) {
			nodeCount += variable.length;
			for (let v of variable)
				if (check(v, depth + 1))
					return true;
		}
		else {
			nodeCount += Object.keys(variable).length;
			for (let key in variable)
				if (check(variable[key], depth + 1))
					return true;
		}
		return false;
	}

	return check(obj, 1);
}

function isTooBigObj(obj, { depthLimit = 100, nodesLimit = 10000, lengthLimit = 1000000 }) {
	let nodeCount = 0;
	let length = 0;

	function check(variable, depth){
		if (depth > depthLimit || nodeCount > nodesLimit || length > lengthLimit)
			return true;
		if (typeof variable === "string")
			length += variable.length;
		else if (typeof variable === "number" || typeof variable === "boolean")
			length += variable.toString().length;
		else if (variable === null)
			length += 4; // "null"
		else if (typeof variable !== "object")
			throw Error("isTooBigObj: unexpected type=" + (typeof variable) + " of " + variable);
		if (length > lengthLimit)
			return true;
		if (typeof variable !== "object" || variable === null)
			return false;
		if (Array.isArray(variable)) {
			nodeCount += variable.length;
			for (let v of variable)
				if (check(v, depth + 1))
					return true;
		}
		else {
			const keys = Object.keys(variable);
			nodeCount += keys.length;
			length += keys.reduce((sum, key) => sum + key.length, 0);
			for (let key in variable)
				if (check(variable[key], depth + 1))
					return true;
		}
		return false;
	}

	return check(obj, 1);
}
