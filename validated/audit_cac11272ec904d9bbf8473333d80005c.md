## Analysis

The attack path described does not exist in this codebase. Two facts rule it out:

1. **`util::from_bytes` is not reachable by unprivileged Move code.** In `aptos-move/framework/aptos-framework/sources/util.move:2-13`, the native is declared `public(friend) native fun from_bytes<T>(bytes: vector<u8>): T;`, and the module's only friends are `aptos_framework::code` and `aptos_framework::gas_schedule` [1](#0-0) . The `account` module never calls `util::from_bytes`, so `native_from_bytes` in `aptos-move/framework/natives/src/util.rs:30-58` is never invoked with attacker-supplied bytes to construct a `RotationProofChallenge` [2](#0-1) .

2. **`RotationProofChallenge` is always constructed on-chain from verified state, never deserialized from user-supplied bytes.** In `account::rotate_authentication_key`, `account::rotate_authentication_key_with_rotation_capability`, and `account::upsert_ed25519_backup_key_on_keyless_account`, the `originator` field is set from `signer::address_of(account)` (the actual transaction signer, or a delegate authorized via a stored capability offer) and `current_auth_key` is derived from the account's on-chain `authentication_key` field via `from_bcs::to_address`, not from attacker-controlled bytes [3](#0-2) [4](#0-3) . The only user-supplied inputs are `cap_rotate_key`/`cap_update_table`, which are raw signature bytes verified against this framework-constructed challenge via `ed25519::signature_verify_strict_t`/`multi_ed25519::signature_verify_strict_t` in `assert_valid_rotation_proof_signature_and_get_auth_key` [5](#0-4) .

Since the challenge struct's `originator`/`current_auth_key` fields are bound by the VM's own signer resolution and on-chain storage — not deserialized via `util::from_bytes`/`native_from_bytes` from caller-supplied `vector<u8>` — there is no path for an unprivileged sender to inject a mismatched `RotationProofChallenge` that corrupts the sender binding used in `rotate_authentication_key`. The `RotationProofChallenge` Rust type in `types/src/account_config/resources/challenge.rs:12-24` is used off-chain (by the CLI and API test helpers, e.g. `crates/aptos/src/account/key_rotation.rs:194-203`) purely to construct the message that gets signed; it is never deserialized on-chain from an attacker-supplied byte vector [6](#0-5) .

fn native_from_bytes(
    context: &mut SafeNativeContext,
    ty_args: &[Type],
    mut args: VecDeque<Value>,
) -> SafeNativeResult<SmallVec<[Value; 1]>> {
    debug_assert_eq!(ty_args.len(), 1);
    debug_assert_eq!(args.len(), 1);

    // TODO(Gas): charge for getting the layout
    let layout = context.type_to_type_layout(&ty_args[0])?;

    let bytes = safely_pop_arg!(args, Vec<u8>);
    context.charge(
        UTIL_FROM_BYTES_BASE + UTIL_FROM_BYTES_PER_BYTE * NumBytes::new(bytes.len() as u64),
    )?;

    let function_value_extension = context.function_value_extension();
    let max_value_nest_depth = context.max_value_nest_depth();
    let val = match ValueSerDeContext::new(max_value_nest_depth)
        .with_legacy_signer()
        .with_func_args_deserialization(&function_value_extension)
        .deserialize(&bytes, &layout)
    {
        Some(val) => val,
        None => return Err(SafeNativeError::abort(EFROM_BYTES)),
    };

    Ok(smallvec![val])
}

    fun assert_valid_rotation_proof_signature_and_get_auth_key(
        scheme: u8,
        public_key_bytes: vector<u8>,
        signature: vector<u8>,
        challenge: &RotationProofChallenge
    ): vector<u8> {
        if (scheme == ED25519_SCHEME) {
            let pk = ed25519::new_unvalidated_public_key_from_bytes(public_key_bytes);
            let sig = ed25519::new_signature_from_bytes(signature);
            assert!(
                ed25519::signature_verify_strict_t(&sig, &pk, *challenge),
                std::error::invalid_argument(EINVALID_PROOF_OF_KNOWLEDGE)
            );
            ed25519::unvalidated_public_key_to_authentication_key(&pk)
        } else if (scheme == MULTI_ED25519_SCHEME) {
            let pk = multi_ed25519::new_unvalidated_public_key_from_bytes(public_key_bytes);
            let sig = multi_ed25519::new_signature_from_bytes(signature);

                std::error::invalid_argument(EINVALID_PROOF_OF_KNOWLEDGE)
            );
            multi_ed25519::unvalidated_public_key_to_authentication_key(&pk)
        } else {
            abort error::invalid_argument(EINVALID_SCHEME)
        }
    }
