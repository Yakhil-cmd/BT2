### Title
Unbounded recursion in `getSourceString`/`getJsonSourceString` reachable via `objectHash.getUnitHash()` before the depth-limit check in `validate()` — potential stack-overflow DoS - (File: string_utils.js)

### Summary
`validation.js`'s `validate()` computes the unit hash of an attacker-supplied joint via `objectHash.getUnitHash(objUnit)` (line 133) **before** it ever calls `isTooDeeplyNestedOrHasTooManyNodes(objUnit)` (line 154), which is the only guard against pathologically deep nesting. `getUnitHash` and related hashing entry points (`getSourceString`, `getJsonSourceString` in `string_utils.js`) recurse into every nested array/object of the unit with no depth limit of their own, relying entirely on the depth check happening downstream.

### Finding Description
`validate()` in `validation.js` performs, in order:
1. `objectHash.getUnitHash(objUnit) !== objUnit.unit` [1](#0-0) 
2. Only afterwards, `isTooDeeplyNestedOrHasTooManyNodes(objUnit)` [2](#0-1) 

`getUnitHash` calls `getStrippedUnit`/`getNakedUnit` and ultimately `getBase64Hash`, which selects `getSourceString` or `getJsonSourceString` depending on unit version [3](#0-2) [4](#0-3) . Both of these string-building functions are naive recursive tree-walkers with **no depth guard**:
- `getSourceString`'s inner `extractComponents` recurses into every array element / object value without any depth counter [5](#0-4) .
- `getJsonSourceString`'s inner `stringify` likewise recurses unconditionally (only de-duplicating via a `WeakMap` for shared references, not limiting depth) [6](#0-5) .

The only defense against deep nesting, `isTooDeeplyNestedOrHasTooManyNodes` (default `depthLimit = 100`), is a *separate* recursive check invoked later in `validate()` [7](#0-6) [2](#0-1) . Because the hash check runs first, a unit whose `messages`/`payload`/nested content contains a deeply nested array or object (e.g., `[[[[...]]]]` at a nesting depth far beyond 100, reaching V8's native call-stack limit) will cause `getSourceString`/`getJsonSourceString` to throw `RangeError: Maximum call stack size exceeded` while still inside the `try { ... } catch(e) { return callbacks.ifJointError("failed to calc unit hash: "+e); } ` block at line 136-138.

This mirrors the CVE-2026-24820 bug class in WickedEngine's Lua (`ldebug.c`): an out-of-bounds/overflow condition triggered by unconstrained recursive stack usage that occurs *before* the intended safety/limit check can reject the malformed input.

### Impact Explanation
Although the immediate `try/catch` around `getUnitHash` at line 131-138 does catch the `RangeError` and turns it into a normal `ifJointError`, this only holds for this exact call site. `getUnitHash`/`getSourceString`/`getJsonSourceString` are also invoked from many other unguarded call sites that compute hashes over attacker-influenced data without a preceding depth check and without a wrapping try/catch in all cases — e.g. `hasValidPayloadHashes` → `getPayloadForHash` → `objectHash.getBase64Hash` per message [8](#0-7) , `getDeviceMessageHashToSign` for device (paired-wallet) messages [9](#0-8) , and `getSignedPackageHashToSign` used for private-payment/signed packages [10](#0-9) . A stack-overflow `RangeError` thrown deep inside these call chains, in a context lacking a matching `try/catch`, crashes the Node.js process (uncaught exception), producing a denial-of-service on the validating node when it processes a single maliciously crafted unit, private payment chain, or paired-device message — i.e., "a network unable to confirm new units" if triggered broadly, or loss of availability for the specific node/wallet handling the message.

### Likelihood Explanation
Any unprivileged unit poster can craft a unit or private-payment/device message containing deeply nested arrays/objects inside fields that reach these hashing functions before nesting depth is validated. Building a sufficiently deep nested structure (thousands of levels) is trivial and requires no special access; the size of the resulting payload is still bounded by ordinary unit-size limits, since JS array/object nesting syntax is compact (`[[[[...]]]]`).

### Recommendation
Move the `isTooDeeplyNestedOrHasTooManyNodes` (and `isObjectWellFormed`) checks to occur strictly before any hashing operation touches attacker-supplied data, in every code path that calls `getUnitHash`, `getBase64Hash`, `getSourceString`, `getJsonSourceString`, `getDeviceMessageHashToSign`, and `getSignedPackageHashToSign` on untrusted input (including message payloads validated individually in `hasValidPayloadHashes`, device messages, and signed packages). Additionally, add an internal depth counter to `extractComponents` in `getSourceString` and to `stringify` in `getJsonSourceString` so these functions fail safely (return an error) instead of relying solely on an external, later-executed guard.

### Proof of Concept
1. Construct a unit joint whose `unit.messages[0].payload` (or another nested field of `objUnit`) contains an array nested far beyond the 100-level default depth limit, e.g. programmatically:
   ```js
   let deep = 1;
   for (let i = 0; i < 50000; i++) deep = [deep];
   ```
2. Embed `deep` inside a message payload of an otherwise well-formed unit, and set `unit.unit` to any 44-char base64 string (the wrong hash is fine — the goal is to trigger the recursion, not pass the hash check).
3. Submit the joint to `validation.js`'s `validate()` (e.g., via a "joint" `justsaying`/unit broadcast). Execution reaches `objectHash.getUnitHash(objUnit)` at `validation.js:133`, which recurses into the deeply nested `deep` array through `getSourceString`/`getJsonSourceString`, throwing `RangeError: Maximum call stack size exceeded` before the `isTooDeeplyNestedOrHasTooManyNodes` check at line 154 ever executes.
4. Repeat the same nested-array payload against a path lacking the protective `try/catch`, such as a device message body handled by `getDeviceMessageHashToSign`, to produce an uncaught exception and crash the process.

*Note: full confirmation of every call site's exception handling (particularly in `network.js`/messaging paths that call `getDeviceMessageHashToSign`/`getSignedPackageHashToSign`) requires deeper tracing than the indexed snippets allow; a live Devin session with full repository access is recommended to verify which of these paths lack a wrapping try/catch and would actually crash the process versus merely returning an error.*

function getJsonSourceString(obj, bAllowEmpty) {
	let cache = new WeakMap();  // object to stringified result
	function stringify(variable){
		if (variable === null)
			throw Error("null value in "+JSON.stringify(obj));
		switch (typeof variable){
			case "string":
				return toWellFormedJsonStringify(variable);
			case "number":
				if (!isFinite(variable))
					throw Error("invalid number: " + variable);
			case "boolean":
				return variable.toString();
			case "object":
				// return cached result if already processed
				if (cache.has(variable))
					return cache.get(variable);
				let result;
				if (Array.isArray(variable)){
					if (variable.length === 0 && !bAllowEmpty)
						throw Error("empty array in "+JSON.stringify(obj));
					result = '[' + variable.map(stringify).join(',') + ']';
				}
				else{
					var keys = Object.keys(variable).sort();
					if (keys.length === 0 && !bAllowEmpty)
						throw Error("empty object in "+JSON.stringify(obj));
					result = '{' + keys.map(function(key){ return toWellFormedJsonStringify(key)+':'+stringify(variable[key]) }).join(',') + '}';
				}
				cache.set(variable, result);  // memoize for future references
				return result;
			default:
				throw Error("getJsonSourceString: unknown type="+(typeof variable)+" of "+variable+", object: "+JSON.stringify(obj));
		}
	}

	return stringify(obj);
}

function isTooDeeplyNestedOrHasTooManyNodes(obj, depthLimit = 100, nodesLimit = 10000) {
	let nodeCount = 0;

	function check(variable, depth){
		if (depth > depthLimit || nodeCount > nodesLimit)
			return true;
		if (variable === null || typeof variable !== "object")
			return false;
		if (Array.isArray(variable)) {
			nodeCount += variable.length;
			for (let v of variable)
				if (check(v, depth + 1))
					return true;
		}
		else {
			nodeCount += Object.keys(variable).length;
			for (let key in variable)
				if (check(variable[key], depth + 1))
					return true;
		}
		return false;
	}

	return check(obj, 1);
}
