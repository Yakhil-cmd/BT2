### Title
Vacuous approval of private-payment chain linkage when only a single element is submitted - (File: network.js)

### Summary
`checkThatEachChainElementIncludesThePrevious` in `network.js` is the function that decides whether a chain of private-payment elements received from a counterparty is a genuine, cryptographically-linked chain before the light wallet accepts and marks it as validated (`linked=1`). When the supplied chain array has exactly one element, the function returns `true` immediately — with no call to `light/get_link_proofs`, no hash verification, and no call into `light.processLinkProofs` — mirroring the nimiq `BlockInclusionProof::is_block_proven` flaw where an empty/degenerate input causes the "proven" check to be skipped entirely and return true.

### Finding Description
`checkThatEachChainElementIncludesThePrevious` is invoked from `updateLinkProofsOfPrivateChain`, which a light wallet uses to verify that a private-payment chain handed to it by a counterparty is authentic before marking `unhandled_private_payments.linked=1`: [1](#0-0) 

The function short-circuits with a hard-coded `return handleResult(true)` whenever `arrPrivateElements.length === 1`, explicitly flagged in the source as `// an issue`: [2](#0-1) 

This is invoked from `updateLinkProofsOfPrivateChain`: [3](#0-2) 

For chains with more than one element, the real verification path is taken: the light vendor is asked for `light/get_link_proofs`, and the response is passed through `light.processLinkProofs`, which cryptographically checks unit hashes, ball hashes, and parent-inclusion relationships between chain elements: [4](#0-3) 

When `arrPrivateElements.length === 1`, none of that verification (unit hash validation, ball-hash validation, or inclusion checking) is performed at all — the single element is trusted outright, exactly analogous to `get_interlink_hops` returning an empty list and causing `is_block_proven` to skip all hash/signature verification and still report "proven."

### Impact Explanation
A malicious private-payment counterparty who constructs and sends a single-element private-payment chain (i.e., presents the very first hop of an asset/output as if it were the entirety of the payment history, omitting all intermediate transfer elements that a genuine multi-hop chain would contain) causes the recipient's light wallet to mark it `linked=1` without any hash or inclusion verification. Depending on how the resulting `linked` payment is subsequently spent or displayed as confirmed, this weakens the wallet's ability to detect a forged or incomplete private-payment provenance, since the single-hop shortcut bypasses the only cryptographic check (`processLinkProofs`) that ties the presented unit into a verifiable, hash-chained history rooted in something the wallet already trusts.

### Likelihood Explanation
Triggering the vacuous-true path requires nothing more than a private-payment counterparty choosing to send an `arrPrivateElements` array of length 1 to a light node peer — a normal, attacker-controlled input with no special preconditions, similar to how the nimiq bug was reachable by simply presenting a target block at a specific (but easily reachable) epoch boundary.

### Recommendation
Remove the length-1 short-circuit and always perform genuine verification of chain-element linkage (or, if a single-element chain is a legitimate degenerate case, explicitly verify that the single element corresponds to the expected root/genesis condition rather than unconditionally returning `true`). At minimum, replace the `// an issue` shortcut with logic that validates the single element's authenticity (e.g., confirming it is genuinely an unlinked first-hop output rather than trusting the counterparty's claim).

### Proof of Concept
1. A malicious private-payment counterparty sends the victim's light wallet a private-payment chain (`arrPrivateElements`) containing only one element, regardless of whether the underlying asset has actually passed through prior hops.
2. The wallet calls `updateLinkProofsOfPrivateChain`, which calls `checkThatEachChainElementIncludesThePrevious`.
3. Because `arrPrivateElements.length === 1`, the function immediately calls `handleResult(true)` without contacting the light vendor for `get_link_proofs` and without invoking `light.processLinkProofs`.
4. `updateLinkProofsOfPrivateChain` treats `bLinked = true`, and the payment row is updated with `linked=1` in `unhandled_private_payments`, despite zero cryptographic verification having taken place. [2](#0-1) [5](#0-4) 

Note: I was unable to fully trace how the `linked` flag downstream governs spending/finality decisions within `private_payment.js` (tool budget exhausted before completing that trace), so the exact severity of consequences after `linked=1` is set could not be fully confirmed from the indexed code and would benefit from further review in a full Devin session.

// light only
// Note that we are leaking to light vendor information about the full chain. 
// If the light vendor was a party to any previous transaction in this chain, he'll know how much we received.
function checkThatEachChainElementIncludesThePrevious(arrPrivateElements, handleResult){
	if (arrPrivateElements.length === 1) // an issue
		return handleResult(true);
	var arrUnits = arrPrivateElements.map(function(objPrivateElement){ return objPrivateElement.unit; });
	requestFromLightVendor('light/get_link_proofs', arrUnits, function(ws, request, response){
		if (!response || response.error)
			return handleResult(null); // undefined result
		var arrChain = response;
		if (!ValidationUtils.isNonemptyArray(arrChain))
			return handleResult(null); // undefined result
		light.processLinkProofs(arrUnits, arrChain, {
			ifError: function(err){
				console.log("linkproof validation failed: "+err);
			//	throw Error(err);
				handleResult(false);
			},
			ifOk: function(){
				console.log("linkproof validated ok");
				handleResult(true);
			}
		});
	});
}

// light only
function updateLinkProofsOfPrivateChain(arrPrivateElements, unit, message_index, output_index, onFailure, onSuccess){
	if (!conf.bLight)
		throw Error("not light but updateLinkProofsOfPrivateChain");
	if (!onFailure)
		onFailure = function(){};
	if (!onSuccess)
		onSuccess = function(){};
	checkThatEachChainElementIncludesThePrevious(arrPrivateElements, function(bLinked){
		if (bLinked === null)
			return onFailure();
		if (!bLinked)
			return deleteHandledPrivateChain(unit, message_index, output_index, onFailure);
		// the result cannot depend on output_index
		db.query("UPDATE unhandled_private_payments SET linked=1 WHERE unit=? AND message_index=?", [unit, message_index], function(){
			onSuccess();
		});
	});
}

function processLinkProofs(arrUnits, arrChain, callbacks){
	try {
		if (!ValidationUtils.isNonemptyArray(arrUnits))
			return callbacks.ifError("invalid units array");
		if (!ValidationUtils.isNonemptyArray(arrChain))
			return callbacks.ifError("invalid chain array");
		if (!arrUnits.every(ValidationUtils.isNonemptyString))
			return callbacks.ifError("units must be non-empty strings");
		if (!arrChain.every(ValidationUtils.isNonemptyObject))
			return callbacks.ifError("chain must be array of non-empty objects");
		// check first element
		var objFirstJoint = arrChain[0];
		if (!ValidationUtils.isNonemptyObject(objFirstJoint.unit))
			return callbacks.ifError("1st element must have unit");
		if (!objFirstJoint || !objFirstJoint.unit || objFirstJoint.unit.unit !== arrUnits[0])
			return callbacks.ifError("unexpected 1st element");
		var assocKnownUnits = {};
		var assocKnownBalls = {};
		assocKnownUnits[arrUnits[0]] = true;
		for (var i = 0; i < arrChain.length; i++) {
			var objElement = arrChain[i];
			if (objElement.unit && objElement.unit.unit) {
				var objJoint = objElement;
				var objUnit = objJoint.unit;
				var unit = objUnit.unit;
				if (!ValidationUtils.isNonemptyString(unit))
					return callbacks.ifError("unit must be non-empty string");
				if (!assocKnownUnits[unit])
					return callbacks.ifError("unknown unit " + unit);
				if (!validation.hasValidHashes(objJoint))
					return callbacks.ifError("invalid hash of unit " + unit);
				assocKnownBalls[objUnit.last_ball] = true;
				assocKnownUnits[objUnit.last_ball_unit] = true;
				if (!ValidationUtils.isNonemptyArray(objUnit.parent_units))
					return callbacks.ifError("no parent_units");
				objUnit.parent_units.forEach(function (parent_unit) {
					assocKnownUnits[parent_unit] = true;
				});
			}
			else if (objElement.unit && objElement.ball) {
				var objBall = objElement;
				if (!ValidationUtils.isNonemptyString(objBall.ball))
					return callbacks.ifError("ball must be non-empty string");
				if (!assocKnownBalls[objBall.ball])
					return callbacks.ifError("unknown ball " + objBall.ball);
				if (objBall.ball !== objectHash.getBallHash(objBall.unit, objBall.parent_balls, objBall.skiplist_balls, objBall.is_nonserial))
					return callbacks.ifError("invalid ball hash");
				if (objBall.unit !== constants.GENESIS_UNIT) {
					if (!ValidationUtils.isNonemptyArray(objBall.parent_balls))
						return callbacks.ifError("no parent_balls");
					objBall.parent_balls.forEach(function (parent_ball) {
						assocKnownBalls[parent_ball] = true;
					});
				}
				if (objBall.skiplist_balls) {
					if (!ValidationUtils.isNonemptyArray(objBall.skiplist_balls))
						return callbacks.ifError("bad skiplist_balls");
					if (!objBall.skiplist_balls.every(ValidationUtils.isNonemptyString))
						return callbacks.ifError("skiplist_balls must be non-empty strings");
					objBall.skiplist_balls.forEach(function (skiplist_ball) {
						assocKnownBalls[skiplist_ball] = true;
					});
				}
				assocKnownUnits[objBall.unit] = true;
			}
			else
				return callbacks.ifError("unrecognized chain element");
		}
		// so, the chain is valid, now check that we can find the requested units in the chain
		for (var i = 1; i < arrUnits.length; i++) // skipped first unit which was already checked
			if (!assocKnownUnits[arrUnits[i]])
				return callbacks.ifError("unit " + arrUnits[i] + " not found in the chain");
	}
	catch (e) {
		return callbacks.ifError("bad chain: " + e.toString());
	}
	callbacks.ifOk();
}
