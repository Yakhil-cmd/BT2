### Title
Malicious Bridge Operator Can Redirect Cross-Chain Value Transfers to an Arbitrary Recipient — (`contracts/service_chain/bridge/BridgeTransferKLAY.sol`, `BridgeTransferERC20.sol`, `BridgeTransferERC721.sol`)

---

### Summary

The `handleKLAYTransfer`, `handleERC20Transfer`, and `handleERC721Transfer` functions in the Kaia service-chain bridge accept a caller-supplied `_to` address with no on-chain binding to the original `RequestValueTransfer` event on the source chain. Because the default operator threshold is **1**, a single malicious operator can unilaterally redirect any in-flight cross-chain transfer to an arbitrary address, draining KLAY or bridged tokens from the destination bridge.

---

### Finding Description

The three handle functions share the same structure:

```solidity
function handleKLAYTransfer(
    bytes32 _requestTxHash,
    address _from,
    address payable _to,   // ← fully caller-controlled
    uint256 _value,
    uint64  _requestedNonce,
    uint64  _requestedBlockNumber,
    bytes memory _extraData
) public onlyOperators nonReentrant {
    _lowerHandleNonceCheck(_requestedNonce);
    if (!_voteValueTransfer(_requestedNonce)) { return; }
    _setHandledRequestTxHash(_requestTxHash);
    ...
    (bool ok, ) = _to.call.value(_value)("");   // ← funds sent to unverified _to
    require(ok, "handleKLAYTransfer: transfer failed");
}
``` [1](#0-0) 

The vote key used in `_voteValueTransfer` is `keccak256(msg.data)`, which commits to all calldata including `_to`. [2](#0-1) 

This means that when threshold ≥ 2, operators must agree on the same `_to` to reach consensus — providing protection. However, the constructor initialises **all** vote-type thresholds to **1**: [3](#0-2) 

With threshold = 1 (the default and the most common deployment configuration), a single operator's call immediately passes the vote and executes the transfer. There is no on-chain mechanism that verifies `_to`, `_value`, or `_from` against the actual source-chain transaction referenced by `_requestTxHash`.

Additionally, `BridgeHandledRequests` only **writes** `handledRequestTx[_requestTxHash] = true` after execution; it is never **read** as a guard inside any handle function, so the mapping provides no replay protection against a different nonce reusing the same hash. [4](#0-3) 

The same unverified-recipient pattern exists in `handleERC20Transfer` (mints or transfers ERC20 to `_to`) and `handleERC721Transfer` (mints or transfers ERC721 to `_to`): [5](#0-4) [6](#0-5) 

---

### Impact Explanation

A malicious operator (or a compromised operator key) with the default threshold of 1 can:

1. Observe a legitimate `RequestValueTransfer` event on the source chain (nonce N, txHash H, value V, legitimate recipient Alice).
2. Call `handleKLAYTransfer(H, from, attacker_address, V, N, blockNum, extraData)`.
3. The single-operator vote passes immediately; KLAY (or ERC20/ERC721) is transferred to `attacker_address` instead of Alice.
4. `closedValueTransferVotes[N]` is set to `true`, permanently preventing the legitimate transfer for nonce N.

The corrupted protected value is the KLAY/ERC20/ERC721 balance of the destination bridge contract: funds are irreversibly sent to the attacker rather than the cross-chain user. The source-chain deposit is already burned/locked, so the user suffers a total loss of the bridged amount.

---

### Likelihood Explanation

- The default threshold is 1, which is the out-of-the-box configuration and is used in all test fixtures.
- The bridge operator account is a hot wallet managed by the service-chain node operator (`node/sc/bridge_manager.go`), making key compromise a realistic threat.
- No cryptographic proof ties `_to` to the source-chain event; the only protection is operator honesty.
- A single operator is sufficient — no collusion is required. [7](#0-6) 

---

### Recommendation

1. **Raise the default threshold** to at least `ceil(N/2) + 1` of registered operators so that no single operator can unilaterally execute a transfer.
2. **Bind `_to` to the source-chain event on-chain**: store a commitment (e.g., `keccak256(txHash, from, to, value, nonce)`) in the `RequestValueTransfer` event and verify it in the handle function, or require operators to submit a Merkle/receipt proof of the source-chain transaction.
3. **Enforce the `handledRequestTx` guard**: add `require(!handledRequestTx[_requestTxHash], "already handled")` at the top of each handle function to prevent the same source-chain transaction from being processed under a different nonce.

---

### Proof of Concept

```
Setup:
  - Bridge deployed with default threshold = 1.
  - Operator = attacker (or attacker compromises the single operator key).
  - Alice sends 100 KLAY on the source chain → RequestValueTransfer(nonce=5, txHash=H, from=Alice, to=Alice_dest, value=100).

Attack:
  attacker.handleKLAYTransfer(
      H,                  // correct txHash (observed from source chain)
      Alice,              // correct _from
      attacker_wallet,    // ← substituted _to
      100,                // correct _value
      5,                  // correct nonce
      blockNum,
      ""
  );

Result:
  - _voteValueTransfer(5) passes immediately (threshold=1, voteKey = keccak256(msg.data) with attacker_wallet).
  - closedValueTransferVotes[5] = true  → Alice's legitimate transfer is permanently blocked.
  - 100 KLAY transferred to attacker_wallet.
  - handledRequestTx[H] = true (informational only, never checked).
```

    function _voteValueTransfer(uint64 _requestNonce)
        internal
        returns(bool)
    {
        require(!closedValueTransferVotes[_requestNonce], "closed vote");

        bytes32 voteKey = keccak256(msg.data);
        if (_voteCommon(VoteType.ValueTransfer, _requestNonce, voteKey)) {
            closedValueTransferVotes[_requestNonce] = true;
            return true;
        }

        return false;
    }

// handleRequestValueTransferEvent handles the given request value transfer event.
func (bi *BridgeInfo) handleRequestValueTransferEvent(ev IRequestValueTransferEvent) error {
	var (
		tokenType                         = ev.GetTokenType()
		tokenAddr, from, to, contractAddr = ev.GetTokenAddress(), ev.GetFrom(), ev.GetTo(), ev.GetRaw().Address
		txHash                            = ev.GetRaw().TxHash
		valueOrTokenId                    = ev.GetValueOrTokenId()
		requestNonce, blkNumber           = ev.GetRequestNonce(), ev.GetRaw().BlockNumber
		extraData                         = ev.GetExtraData()
	)

	ctpartTokenAddr := bi.GetCounterPartToken(tokenAddr)
	// TODO-Kaia-Servicechain Add counterpart token address in requestValueTransferEvent
	if tokenType != KAIA && ctpartTokenAddr == (common.Address{}) {
		logger.Warn("Unregistered counter part token address.", "addr", ctpartTokenAddr.Hex())
		ctTokenAddr, err := bi.counterpartBridge.RegisteredTokens(nil, tokenAddr)
		if err != nil {
			return err
		}
		if ctTokenAddr == (common.Address{}) {
			return errors.New("can't get counterpart token from bridge")
		}
		if err := bi.RegisterToken(tokenAddr, ctTokenAddr); err != nil {
			return err
		}
		ctpartTokenAddr = ctTokenAddr
		logger.Info("Register counter part token address.", "addr", ctpartTokenAddr.Hex(), "cpAddr", ctTokenAddr.Hex())
	}

	bridgeAcc := bi.account

	bridgeAcc.Lock()
	defer bridgeAcc.UnLock()

	auth := bridgeAcc.GenerateTransactOpts()

	var handleTx *types.Transaction
	var err error

	switch tokenType {
	case KAIA:
		handleTx, err = bi.bridge.HandleKLAYTransfer(auth, txHash, from, to, valueOrTokenId, requestNonce, blkNumber, extraData)
		if err != nil {
			return err
		}
		handleValueTransferLog(bi.onChildChain, handleVTmethods[KAIA], handleTx.Hash().String(), requestNonce, from, to, valueOrTokenId)
	case ERC20:
		handleTx, err = bi.bridge.HandleERC20Transfer(auth, txHash, from, to, ctpartTokenAddr, valueOrTokenId, requestNonce, blkNumber, extraData)
		if err != nil {
			return err
		}
		handleValueTransferLog(bi.onChildChain, handleVTmethods[ERC20], handleTx.Hash().String(), requestNonce, from, to, valueOrTokenId)
	case ERC721:
		uri := GetURI(ev)
		handleTx, err = bi.bridge.HandleERC721Transfer(auth, txHash, from, to, ctpartTokenAddr, valueOrTokenId, requestNonce, blkNumber, uri, extraData)
		if err != nil {
			return err
		}
		handleValueTransferLog(bi.onChildChain, handleVTmethods[ERC721], handleTx.Hash().String(), requestNonce, from, to, valueOrTokenId)
	default:
		logger.Error("Got Unknown Token Type ReceivedEvent", "bridge", contractAddr, "nonce", requestNonce, "from", from)
		return nil
	}

	bridgeAcc.IncNonce()

	bi.bridgeDB.WriteHandleTxHashFromRequestTxHash(txHash, handleTx.Hash())
	return nil
}
