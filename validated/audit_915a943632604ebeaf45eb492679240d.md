### Title
Path traversal via unsanitized release asset `Name` in `gh release download` allows arbitrary file write - ([File: pkg/cmd/release/download/download.go])

### Summary
`downloadRun` builds download targets directly from the attacker-controlled `ReleaseAsset.Name` field returned by the GitHub API, and passes it unmodified through `downloadAsset` into `destinationWriter.Copy`/`makePath`, which only does `filepath.Join(w.dir, name)` with no path-traversal check. A release asset whose `Name` contains `../` segments (e.g. `../../evil.sh`) can therefore cause the downloaded file to be written outside `opts.Destination`.

### Finding Description
`shared.ReleaseAsset.Name` is decoded straight from the release API JSON with no validation [1](#0-0) . In `downloadRun`, this `a.Name` becomes `downloadTarget.name` and is used verbatim, whether or not `--pattern` is supplied [2](#0-1) . It then flows into `downloadAsset(dest, httpClient, a.url, a.name, isArchive)` as `fileName`, and since `fileName` is non-empty, the Content-Disposition parsing branch (which does apply `filepath.Base`) is skipped entirely [3](#0-2) . The raw name is passed to `dest.Copy(fileName, resp.Body)`, which calls `makePath(name)` — simply `filepath.Join(w.dir, name)` — with no confinement check against `w.dir`, before `os.OpenFile(fp, os.O_WRONLY|os.O_CREATE|os.O_TRUNC, 0644)` [4](#0-3) . `filepath.Join`/`filepath.Clean` do not prevent `..` from escaping the base directory; they merely normalize the path lexically. Only the archive-mode branch (`fileName == ""`, i.e. `--archive` download whose name comes from `Content-Disposition`) sanitizes the name with `filepath.Base(serverFileName)` [5](#0-4) ; the asset-name path used for ordinary/`--pattern` downloads has no equivalent protection.

The Windows-reserved-filename check only inspects the base name and does nothing to block `..` traversal [6](#0-5) . No other sanitization (safepaths, filepath.Base, or explicit `..` rejection) exists anywhere on this code path for the asset `Name`.

### Impact Explanation
An attacker who can publish a release on any public repository (a normal, unprivileged GitHub action) can name an asset something like `../../.bashrc` or `../../../.ssh/authorized_keys`. When a victim runs `gh release download` (with no `--pattern`, or with a `--pattern` that matches the crafted name, e.g. `*`), `os.OpenFile` will create/truncate/overwrite a file at an attacker-chosen path relative to the destination directory, anywhere the victim's OS user has write permission. This is an arbitrary file write / overwrite outside the intended destination directory, matching GitHub's file write/overwrite outside intended path impact class.

### Likelihood Explanation
Feasibility depends on whether the GitHub Releases API itself accepts and returns asset names containing `/` or `..` sequences uploaded by an unprivileged user. That server-side constraint is external to this repository and could not be verified from the codebase alone. If the API permits such names (or if a compatible/enterprise/GHES release API endpoint does), the client-side exploitation described is fully reachable and repeatable for every `gh release download` invocation against the malicious release, with no additional victim interaction required beyond running the ordinary command.

### Recommendation
Sanitize `name` the same way for all code paths before joining with the destination directory: apply `filepath.Base` (as already done for the Content-Disposition-derived archive name) or explicitly reject/strip any path separators and `..` segments in `destinationWriter.makePath`, and additionally verify with `filepath.Rel`/prefix-checking that the resulting `fp` remains within `filepath.Clean(w.dir)` before opening the file, returning an error otherwise.

    // is written strictly under tmpDest.
    require.NoError(t, err)

func downloadAsset(dest *destinationWriter, httpClient *http.Client, assetURL safeurl.SafeURL, fileName string, isArchive bool) error {
	if err := dest.Check(fileName); err != nil {
		return err
	}

	req, err := http.NewRequest("GET", assetURL.String(), nil)
	if err != nil {
		return err
	}

	req.Header.Set("Accept", "application/octet-stream")
	if isArchive {
		// adding application/json to Accept header due to a bug in the zipball/tarball API endpoint that makes it mandatory
		req.Header.Set("Accept", "application/octet-stream, application/json")

		// override HTTP redirect logic to avoid "legacy" Codeload resources
		oldClient := *httpClient
		httpClient = &oldClient
		httpClient.CheckRedirect = func(req *http.Request, via []*http.Request) error {
			if len(via) == 1 {
				req.URL.Path = removeLegacyFromCodeloadPath(req.URL.Path)
			}
			return nil
		}
	}

	resp, err := httpClient.Do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()

	if resp.StatusCode > 299 {
		return api.HandleHTTPError(resp)
	}

	if len(fileName) == 0 {
		contentDisposition := resp.Header.Get("Content-Disposition")

		_, params, err := mime.ParseMediaType(contentDisposition)
		if err != nil {
			return fmt.Errorf("unable to parse file name of archive: %w", err)
		}
		if serverFileName, ok := params["filename"]; ok {
			fileName = filepath.Base(serverFileName)
		} else {
			return errors.New("unable to determine file name of archive")
		}
	}

	return dest.Copy(fileName, resp.Body)

func (w destinationWriter) makePath(name string) string {
	if w.file == "" {
		return filepath.Join(w.dir, name)
	}
	return w.file
}

// Check returns an error if a file already exists at destination
func (w destinationWriter) Check(name string) error {
	if name == "" {
		// skip check as file name will only be known after the API request
		return nil
	}
	fp := w.makePath(name)
	if fp == "-" {
		// writing to stdout should always proceed
		return nil
	}
	return w.check(fp)
}

func (w destinationWriter) check(fp string) error {
	if _, err := os.Stat(fp); err == nil {
		if w.skipExisting {
			return errSkipped
		}
		if !w.overwrite {
			return fmt.Errorf(
				"%s already exists (use `--clobber` to overwrite file or `--skip-existing` to skip file)",
				fp,
			)
		}
	}
	return nil
}

// Copy writes the data from r into a file specified by name.
func (w destinationWriter) Copy(name string, r io.Reader) (copyErr error) {
	fp := w.makePath(name)
	if fp == "-" {
		if w.allowEscapes {
			_, copyErr = io.Copy(w.stdout, r)
			return
		}
		copyErr = iostreams.CopyGuardedContent(w.stdout, r, w.isTTY)
		if binErr, ok := errors.AsType[iostreams.BinaryTerminalError](copyErr); ok {
			copyErr = fmt.Errorf("%w; use `--output` to save it to a file, or pass --allow-escape-sequences to output it anyway", binErr)
		} else if errors.Is(copyErr, iostreams.ErrEscapeSequence) {
			copyErr = errors.New("the asset contains terminal escape sequences; use `--output` to save it to a file, or pass --allow-escape-sequences to output it anyway")
		}
		return
	}
	if copyErr = w.check(fp); copyErr != nil {
		return
	}

	if dir := filepath.Dir(fp); dir != "." {
		if copyErr = os.MkdirAll(dir, 0755); copyErr != nil {
			return
		}
	}

	var f *os.File
	if f, copyErr = os.OpenFile(fp, os.O_WRONLY|os.O_CREATE|os.O_TRUNC, 0644); copyErr != nil {
		return
	}

	defer func() {
		if err := f.Close(); copyErr == nil && err != nil {
			copyErr = err
		}
	}()

	_, copyErr = io.Copy(f, r)
	return
}

func isWindowsReservedFilename(filename string) bool {
	// Windows terminals should prevent the creation of these files
	// but that behavior is not enforced across terminals. Prevent
	// the user from downloading files with these reserved names as
	// they represent an exploit vector for bad actors.
	// Reserved filenames defined at:
	// https://learn.microsoft.com/en-us/windows/win32/fileio/naming-a-file#win32-file-namespaces
	reservedFilenames := []string{"CON", "PRN", "AUX", "NUL", "COM0",
		"COM1", "COM2", "COM3", "COM4", "COM5",
		"COM6", "COM7", "COM8", "COM9", "COM¹",
		"COM²", "COM³", "LPT0", "LPT1", "LPT2",
		"LPT3", "LPT4", "LPT5", "LPT6", "LPT7",
		"LPT8", "LPT9", "LPT¹", "LPT²", "LPT³"}

	// Normalize type case and remove file type extension from filename.
	filename = strings.ToUpper(strings.Split(filename, ".")[0])

	return slices.Contains(reservedFilenames, filename)
}
