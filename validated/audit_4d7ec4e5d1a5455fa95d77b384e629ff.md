- `validation.js:236` calls `objectHash.getBase64Hash(getPayloadForHash(m), ...)` → `getBase64Hash` → `getSourceString` (`object_hash.js:27-29`) to verify the `payload_hash` of *every* inline message, for *every* posted unit.

- The `data`/`profile` case in `validateInlinePayload` (`validation.js:1954-1965`) only checks `typeof payload === "object"` — it does **not** call `isTooDeeplyNestedOrHasTooManyNodes`, unlike the analogous device-message path in `network.js:3520` (`isTooDeeplyNestedOrHasTooManyNodes(objDeviceMessage, 5, 100)`).

### Title
Uncontrolled recursion / stack overflow via deeply nested `data`/`profile` message payload in unit validation - (File: string_utils.js, object_length.js)

### Summary
`validation.js` computes each message's `payload_hash` and the unit's `headers_commission`/`payload_commission` for every incoming unit by calling `object_hash.getBase64Hash()`/`getUnitContentHash()` and `object_length.getHeadersSize()`/`getTotalPayloadSize()`. Both ultimately recurse into the payload structure with `extractComponents()` (`string_utils.js`) and `_getLength()` (`object_length.js`), which have no recursion-depth guard. Unlike device messages (`network.js`, guarded by `isTooDeeplyNestedOrHasTooManyNodes`) or AA definitions/oscript formulas (guarded by `MAX_DEPTH`/`depth>100` checks in `aa_validation.js` and `formula/validation.js`), the free-form `data`/`profile`/`data_feed` message payloads accepted in ordinary (non-AA) units are validated in `validateInlinePayload` with only a shallow `typeof payload === "object"` check before the unguarded hash/length routines already ran on them.

### Finding Description
Any unprivileged node can broadcast a unit containing a `data` (or `profile`/`data_feed`) message whose `payload` is a deeply nested JSON structure (e.g., thousands of nested arrays). During basic joint/unit validation, before any application-specific depth checks fire, the code path `validate()` → payload-hash verification (`objectHash.getBase64Hash`) and `getHeadersSize`/`getTotalPayloadSize` (`objectLength.getLength`) will recursively walk the entire nested structure via `extractComponents`/`_getLength`, one JS stack frame per nesting level, with no cap. A sufficiently deep payload (achievable within `MAX_UNIT_LENGTH`/message size limits since deep nesting requires very little serialized bytes, e.g. `[[[[...1...]]]]`) exhausts the V8 call stack and crashes the validating process with `RangeError: Maximum call stack size exceeded`, before the process even reaches `validateInlinePayload`'s type check.

### Impact Explanation
Any full node (including witnesses, hubs, and light-serving nodes) that validates a maliciously crafted unit crashes due to stack exhaustion. Since every node in the network must independently validate every unit that reaches it via normal, non-privileged posting, a single attacker-controlled unit can be used to repeatedly crash nodes attempting to validate/relay it, disrupting the network's ability to confirm new units and reach consensus — matching the "network unable to confirm new units" acceptance criterion.

### Likelihood Explanation
Likelihood is high: constructing a deeply nested JSON array as a `data` message payload requires no special privileges, no signatures beyond a normal unit author, and is trivial to script. The bug is triggered purely by the structure of an otherwise well-formed message during standard validation of any freshly received unit.

### Recommendation
Add the same `isTooDeeplyNestedOrHasTooManyNodes` (or equivalent bounded, depth-limited/iterative) check used for device messages to `data`, `profile`, and `data_feed` payloads (and any other free-form payload) in `validateInlinePayload`, and apply it *before* `getBase64Hash`/`getHeadersSize`/`getTotalPayloadSize` are invoked on message payloads in `validate()`. Alternatively/additionally, convert `extractComponents()` in `string_utils.js` and `_getLength()` in `object_length.js` to iterative (stack-based) traversal or enforce an explicit maximum recursion depth consistent with the limits already used elsewhere in the codebase (e.g., depth 100 as in `formula/validation.js`).

### Proof of Concept
1. Construct a unit with a `data` message whose payload is deeply nested, e.g. programmatically build `payload = 1; for (i=0;i<50000;i++) payload = [payload];` (an array nested 50,000 levels deep), keeping serialized size within `MAX_UNIT_LENGTH`.
2. Set `payload_hash` to any placeholder — validation will call `objectHash.getBase64Hash(getPayloadForHash(m))` on the real payload to check integrity, which invokes `getSourceString` → `extractComponents`, recursing 50,000 stack frames deep before ever comparing hashes.
3. Broadcast the joint to a node running `ocore` (or post via `network.js` handling of incoming units, which ultimately calls `validation.js`'s `validate()`).
4. The receiving node's Node.js process throws `RangeError: Maximum call stack size exceeded` inside `extractComponents`/`_getLength`, crashing (or hanging, depending on process supervision) the validating node.

function getSourceString(obj) {
	var arrComponents = [];
	function extractComponents(variable){
		if (variable === null)
			throw Error("null value in "+JSON.stringify(obj));
		switch (typeof variable){
			case "string":
				if (variable.includes(STRING_JOIN_CHAR))
					throw Error("00 byte in string value in " + JSON.stringify(obj));
				arrComponents.push("s", variable);
				break;
			case "number":
				if (!isFinite(variable))
					throw Error("invalid number: " + variable);
				arrComponents.push("n", variable.toString());
				break;
			case "boolean":
				arrComponents.push("b", variable.toString());
				break;
			case "object":
				if (Array.isArray(variable)){
					if (variable.length === 0)
						throw Error("empty array in "+JSON.stringify(obj));
					arrComponents.push('[');
					for (var i=0; i<variable.length; i++)
						extractComponents(variable[i]);
					arrComponents.push(']');
				}
				else{
					var keys = Object.keys(variable).sort();
					if (keys.length === 0)
						throw Error("empty object in "+JSON.stringify(obj));
					keys.forEach(function(key){
						if (typeof variable[key] === "undefined")
							throw Error("undefined at "+key+" of "+JSON.stringify(obj));
						if (key.includes(STRING_JOIN_CHAR))
							throw Error("00 byte in object key in " + JSON.stringify(obj));
						arrComponents.push(key);
						extractComponents(variable[key]);
					});
				}
				break;
			default:
				throw Error("getSourceString: unknown type="+(typeof variable)+" of "+variable+", object: "+JSON.stringify(obj));
		}
	}

	extractComponents(obj);
	return arrComponents.join(STRING_JOIN_CHAR);
}

function getLength(value, bWithKeys) {
	let cache = new WeakMap();  // object to length
	function _getLength(value) {
		if (value === null)
			return 0;
		switch (typeof value) {
			case "string":
				return value.length;
			case "number":
				if (!isFinite(value))
					throw Error("invalid number: " + value);
				return 8;
				//return value.toString().length;
			case "object":
				// return cached result if already processed
				if (cache.has(value))
					return cache.get(value);
				var len = 0;
				if (Array.isArray(value))
					value.forEach(function (element) {
						len += _getLength(element);
					});
				else
					for (var key in value) {
						if (!Object.prototype.hasOwnProperty.call(value, key))
							throw Error("object has non-own property " + key + " in " + JSON.stringify(value));
						if (typeof value[key] === "undefined")
							throw Error("undefined at " + key + " of " + JSON.stringify(value));
						if (bWithKeys)
							len += key.length;
						len += _getLength(value[key]);
					}
				cache.set(value, len);  // memoize for future references
				return len;
			case "boolean":
				return 1;
			default:
				throw Error("unknown type=" + (typeof value) + " of " + value);
		}
	}
	return _getLength(value);
}
