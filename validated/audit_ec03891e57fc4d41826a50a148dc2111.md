## Title
Uncontrolled Resource Consumption via Unbounded Private Payment Chains from Paired Devices - (File: `wallet.js`)

### Summary
`wallet.js`'s `handlePrivatePaymentChains()` handles the `private_payment_chains` message that a paired device (or, when relayed, another private-payment counterparty) can send through the hub. The function validates only that each chain and its elements are *non-empty* arrays/objects — it never bounds the number of chains, the length of each chain, or the size/complexity of `inputs`/`outputs` inside each chain element before hashing, DB-querying, validating and recursively forwarding every element.

### Finding Description
`handlePrivatePaymentChains(ws, body, from_address, callbacks)` accepts `body.chains` and only checks shape, not size: [1](#0-0) 

There is no cap on `arrChains.length`, on the length of each individual chain (`c.length`), or on `e.payload.inputs`/`e.payload.outputs` array sizes. Each chain is then run through `async.eachSeries`, computing an `objectHash.getBase64Hash` per head element and calling `network.handleOnlinePrivatePayment` for every chain: [2](#0-1) 

`handleOnlinePrivatePayment` in turn drives `privatePayment.validateAndSavePrivatePaymentChain`, which walks the entire chain element-by-element (`parsePrivatePaymentChain` in `indivisible_asset.js` / the divisible variant), issuing DB queries, spend-proof hash computations, and `_.cloneDeep` on the whole chain for every element: [3](#0-2) [4](#0-3) 

After validation, the same unbounded `arrChains` is fanned out again to every wallet member sharing an output address via `forwardPrivateChainsToOtherMembersOfOutputAddresses`, multiplying the cost across paired devices: [5](#0-4) 

Unlike unit/AA/definition validation elsewhere in the codebase — which enforce `MAX_DEPTH`, `MAX_COMPLEXITY`, `MAX_OPS`, and explicit array-size caps (e.g. `MAX_DENOMINATIONS_PER_ASSET_DEFINITION`, `weighted and` `args.set` limits, `too many denominations`) — this device-message code path has no equivalent ceiling on chain count, chain depth, or per-element array sizes.

### Impact Explanation
A paired device (a normal, unprivileged correspondent in the wallet-to-wallet protocol) can send a single `private_payment_chains` message containing an arbitrarily large number of chains, or chains with an arbitrarily large number of elements/inputs/outputs. Processing this triggers unbounded hashing, `cloneDeep`, DB queries, and recursive forwarding to every other device sharing the output addresses, consuming excessive CPU/memory on the receiving wallet and any downstream device it forwards to — a denial-of-service against the victim wallet/hub, mirroring the jose2go uncontrolled-resource-consumption bug class (unbounded resource use from unvalidated structured input).

### Likelihood Explanation
Low-to-medium: it requires the attacker to be a paired device or a private-payment counterparty able to reach the wallet's hub-message handler, but no special privileges, signatures, or prior trust beyond pairing are required to send an oversized `chains` payload.

### Recommendation
Add explicit bounds in `handlePrivatePaymentChains` (and the underlying `parsePrivatePaymentChain`/`validatePrivatePayment` code paths) analogous to the limits already used elsewhere (`MAX_DEPTH`, `MAX_OPS`-style checks): cap `arrChains.length`, cap each chain's length, and cap `payload.inputs`/`payload.outputs` array sizes before any hashing, cloning, or DB work is performed.

### Proof of Concept
1. Pair a malicious device with a victim wallet (normal pairing flow, no special privilege).
2. Send a `private_payment_chains` message via the hub with `body.chains` containing, e.g., tens of thousands of synthetic chains (or a small number of chains each with a very long array of `inputs`/`outputs`), each satisfying only the minimal shape checks in lines 959–972 of `wallet.js`.
3. Observe the victim wallet perform `objectHash.getBase64Hash` and `_.cloneDeep` per chain/element, issue a DB query per chain element, and then forward the entire (still oversized) `arrChains` array to every other paired device sharing an output address — consuming excessive CPU/memory and potentially stalling the wallet process.

function parsePrivatePaymentChain(conn, arrPrivateElements, callbacks){
	var bAllStable = true;
	var issuePrivateElement = arrPrivateElements[arrPrivateElements.length-1];
	if (!issuePrivateElement.payload || !issuePrivateElement.payload.inputs || !issuePrivateElement.payload.inputs[0])
		return callbacks.ifError("invalid issue private element");
	var asset = issuePrivateElement.payload.asset;
	if (!asset)
		return callbacks.ifError("no asset in issue private element");
	var denomination = issuePrivateElement.payload.denomination;
	if (!denomination)
		return callbacks.ifError("no denomination in issue private element");
	async.forEachOfSeries(
		arrPrivateElements,
		function(objPrivateElement, i, cb){
			if (!objPrivateElement.payload || !objPrivateElement.payload.inputs || !objPrivateElement.payload.inputs[0])
				return cb("invalid payload");
			if (!objPrivateElement.output)
				return cb("no output in private element");
			if (objPrivateElement.payload.asset !== asset)
				return cb("private element has a different asset");
			if (objPrivateElement.payload.denomination !== denomination)
				return cb("private element has a different denomination");
			var prevElement = null; 
			if (i+1 < arrPrivateElements.length){ // excluding issue transaction
				var prevElement = arrPrivateElements[i+1];
				if (prevElement.unit !== objPrivateElement.payload.inputs[0].unit)

// handles one private payload and its chain
function handleOnlinePrivatePayment(ws, arrPrivateElements, bViaHub, callbacks){
	if (!ValidationUtils.isNonemptyArray(arrPrivateElements))
		return callbacks.ifError("private_payment content must be non-empty array");
	
	var unit = arrPrivateElements[0].unit;
	var message_index = arrPrivateElements[0].message_index;
	var output_index = arrPrivateElements[0].payload.denomination ? arrPrivateElements[0].output_index : -1;
	if (!ValidationUtils.isValidBase64(unit, constants.HASH_LENGTH))
		return callbacks.ifError("invalid unit");
	if (!ValidationUtils.isNonnegativeInteger(message_index))
		return callbacks.ifError("invalid message_index");
	if (!(ValidationUtils.isNonnegativeInteger(output_index) || output_index === -1))
		return callbacks.ifError("invalid output_index");

	var savePrivatePayment = function(cb){
		// we may receive the same unit and message index but different output indexes if recipient and cosigner are on the same device.
		// in this case, we also receive the same (unit, message_index, output_index) twice - as cosigner and as recipient.  That's why IGNORE.
		db.query(
			"INSERT "+db.getIgnore()+" INTO unhandled_private_payments (unit, message_index, output_index, json, peer) VALUES (?,?,?,?,?)", 
			[unit, message_index, output_index, JSON.stringify(arrPrivateElements), bViaHub ? '' : ws.peer], // forget peer if received via hub
			function(){
				callbacks.ifQueued();
				if (cb)
					cb();
			}
		);
	};
	
	if (conf.bLight && arrPrivateElements.length > 1){
		savePrivatePayment(function(){
			updateLinkProofsOfPrivateChain(arrPrivateElements, unit, message_index, output_index);
			rerequestLostJointsOfPrivatePayments(); // will request the head element
		});
		return;
	}

	joint_storage.checkIfNewUnit(unit, {
		ifKnown: function(){
			//assocUnitsInWork[unit] = true;
			privatePayment.validateAndSavePrivatePaymentChain(arrPrivateElements, {
				ifOk: function(){
					//delete assocUnitsInWork[unit];
					callbacks.ifAccepted(unit);
					eventBus.emit("new_my_transactions", [unit]);
				},
				ifError: function(error){
					//delete assocUnitsInWork[unit];
					callbacks.ifValidationError(unit, error);
				},
				ifWaitingForChain: function(){
					savePrivatePayment();
				}
			});
		},
		ifNew: function(){
			savePrivatePayment();
			// if received via hub, I'm requesting from the same hub, thus telling the hub that this unit contains a private payment for me.
			// It would be better to request missing joints from somebody else
			requestNewMissingJoints(ws, [unit]);
		},
		ifKnownUnverified: savePrivatePayment,
		ifKnownBad: function(){
			callbacks.ifValidationError(unit, "known bad");
		}
	});
}
