### Title
Stack exhaustion via deeply nested unit content processed by unbounded recursive hashing before depth check - (File: validation.js, object_hash.js, string_utils.js)

### Summary
`validate()` in `validation.js` computes the unit's hash by calling `objectHash.getUnitHash(objUnit)` immediately after checking the unit's basic hash length, and only afterward calls `isTooDeeplyNestedOrHasTooManyNodes(objUnit)` to reject overly nested/oversized units. Because the hash-computation path recurses over the *entire* unit content (including message payloads) with no depth limit, an attacker-crafted unit with an extremely deeply nested payload can exhaust the call stack before the depth-limiting guard ever executes.

### Finding Description
In `validation.js`, the `validate(objJoint, callbacks, external_conn)` function performs:
```js
if (objectHash.getUnitHash(objUnit) !== objUnit.unit)
    return callbacks.ifJointError(...);
...
if (isTooDeeplyNestedOrHasTooManyNodes(objUnit))
    return ... "unit is too deeply nested" ...
``` [1](#0-0) 

`objectHash.getUnitHash()` calls `getStrippedUnit()`/`getNakedUnit()` (which does a full `_.cloneDeep(objUnit)` of the entire unit, including message payloads) and ultimately `getBase64Hash()` → `getSourceString(obj)` in `string_utils.js`. [2](#0-1) 

`getSourceString()` implements an **unbounded** recursive inner function `extractComponents(variable)` that walks every array element and object key with no depth ceiling and no periodic call-stack interruption (no `setImmediate`/iteration cap), unlike other recursive walkers in the codebase:
```js
function extractComponents(variable){
    ...
    case "object":
        if (Array.isArray(variable)){
            ...
            for (var i=0; i<variable.length; i++)
                extractComponents(variable[i]);
            ...
        }
        else{
            ...
            keys.forEach(function(key){
                ...
                extractComponents(variable[key]);
            });
        }
        break;
``` [3](#0-2) 

This is precisely the bug class described in the MongoDB report: a recursive function that processes deeply nested documents without periodically checking/limiting depth. The codebase elsewhere clearly recognizes this risk and defends against it — e.g. `isTooDeeplyNestedOrHasTooManyNodes()` (depth/node limit 100/10000) is explicitly used to reject deeply nested units, `formula/validation.js`'s `evaluate()` checks `depth > 100`, and `aa_validation.js` enforces `MAX_DEPTH = 100` with `setImmediate` interruption every 100 iterations. [4](#0-3) [5](#0-4) [6](#0-5) 

However, that protective check (`isTooDeeplyNestedOrHasTooManyNodes`) at `validation.js:154` runs strictly *after* `objectHash.getUnitHash(objUnit)` at `validation.js:133`, which itself triggers the unbounded recursion (via `_.cloneDeep` in `getNakedUnit` and `extractComponents` in `getSourceString`) over the full, unvalidated unit content. Any unprivileged peer can post a unit/joint (via `network.js`'s `'joint'` handler, which itself calls `isTooDeeplyNestedOrHasTooManyNodes` on the joint only after receiving it, but the light/full validation flow ultimately reaches `validate()`) containing a message (e.g., a `data` or `data_feed` payload, or a nested `definition_template`) with thousands of nested arrays/objects, causing the hash-verification step to overflow the stack before any size/depth rejection occurs.

### Impact Explanation
An unbounded stack overflow during mandatory unit-hash verification crashes the Node.js process handling validation (full node or light vendor), since JavaScript `RangeError: Maximum call stack size exceeded` from deep recursion is not reliably catchable within `try { ... } catch(e)` boundaries under V8 in all call configurations (and even if caught, since it occurs at `validate()`'s outer `try` block around the hash check, it may propagate through worker/mutex logic and abort the process or corrupt in-flight state). This allows a single crafted unit posted by any unprivileged unit poster to crash validating nodes repeatedly, preventing the network from confirming new units and constituting a denial-of-service condition against consensus-critical validation logic — directly analogous to the reported mongod behavior of running out of stack memory while evaluating expressions that produce deeply nested documents.

### Likelihood Explanation
The path is reachable by any unit author with no special privileges: crafting a JSON payload (e.g., for a `text`/`data`/`data_feed` message app) with deep array/object nesting is trivial and requires no valid signatures beyond what's needed to get to the hash-check stage, since the ordering places `getUnitHash()` before the depth guard. Likelihood is high given the low barrier to construct such a payload and the wide network exposure of unit/joint validation.

### Recommendation
Reorder the checks in `validate()` in `validation.js` so `isTooDeeplyNestedOrHasTooManyNodes(objUnit)` (and `isObjectWellFormed`) execute *before* `objectHash.getUnitHash(objUnit)` is invoked, ensuring no unbounded recursive traversal (hashing, cloning) occurs on unvalidated, potentially deeply nested content. Additionally, harden `getSourceString()`/`extractComponents()` in `string_utils.js` and `_.cloneDeep` usage in `object_hash.js`'s `getNakedUnit()` with an explicit recursion-depth limit consistent with the `depthLimit`/`nodesLimit` used elsewhere in the codebase, so that even out-of-order invocations cannot cause unbounded stack growth.

### Proof of Concept
1. Construct a unit joint whose `unit.messages` includes a message (e.g., `app: "data"`) with a `payload` consisting of deeply nested arrays, for example an object built from the pattern `'['.repeat(100000) + '1' + ']'.repeat(100000)` parsed as JSON (or the equivalent nested-object form), assembled as a valid-looking unit structure with correct `unit.unit` field omitted/guessed so that `validate()` proceeds to compute `objectHash.getUnitHash(objUnit)`.
2. Submit this joint to a node via the `'joint'` justsaying handler in `network.js`, or via light-vendor endpoints that eventually call `validation.js`'s `validate()`.
3. Observe that `getUnitHash()` → `getNakedUnit()`/`getStrippedUnit()` → `getSourceString()`'s recursive `extractComponents()` walks the deeply nested payload before `isTooDeeplyNestedOrHasTooManyNodes()` can reject it, causing `RangeError: Maximum call stack size exceeded` and process instability/crash, matching the existing test pattern in `test/formula.test.js` (`'deeply nested array'`/`'deeply nested dictionary'` tests at depth 2000) but applied to the unit-hash-verification code path which lacks equivalent depth protection. function getSourceString(obj) {
	var arrComponents = [];
	function extractComponents(variable){
		if (variable === null)
			throw Error("null value in "+JSON.stringify(obj));
		switch (typeof variable){
			case "string":
				if (variable.includes(STRING_JOIN_CHAR))
					throw Error("00 byte in string value in " + JSON.stringify(obj));
				arrComponents.push("s", variable);
				break;
			case "number":
				if (!isFinite(variable))
					throw Error("invalid number: " + variable);
				arrComponents.push("n", variable.toString());
				break;
			case "boolean":
				arrComponents.push("b", variable.toString());
				break;
			case "object":
				if (Array.isArray(variable)){
					if (variable.length === 0)
						throw Error("empty array in "+JSON.stringify(obj));
					arrComponents.push('[');
					for (var i=0; i<variable.length; i++)
						extractComponents(variable[i]);
					arrComponents.push(']');
				}
				else{
					var keys = Object.keys(variable).sort();
					if (keys.length === 0)
						throw Error("empty object in "+JSON.stringify(obj));
					keys.forEach(function(key){
						if (typeof variable[key] === "undefined")
							throw Error("undefined at "+key+" of "+JSON.stringify(obj));
						if (key.includes(STRING_JOIN_CHAR))
							throw Error("00 byte in object key in " + JSON.stringify(obj));
						arrComponents.push(key);
						extractComponents(variable[key]);
					});
				}
				break;
			default:
				throw Error("getSourceString: unknown type="+(typeof variable)+" of "+variable+", object: "+JSON.stringify(obj));
		}
	}

	extractComponents(obj);
	return arrComponents.join(STRING_JOIN_CHAR);
}

function isTooDeeplyNestedOrHasTooManyNodes(obj, depthLimit = 100, nodesLimit = 10000) {
	let nodeCount = 0;

	function check(variable, depth){
		if (depth > depthLimit || nodeCount > nodesLimit)
			return true;
		if (variable === null || typeof variable !== "object")
			return false;
		if (Array.isArray(variable)) {
			nodeCount += variable.length;
			for (let v of variable)
				if (check(v, depth + 1))
					return true;
		}
		else {
			nodeCount += Object.keys(variable).length;
			for (let key in variable)
				if (check(variable[key], depth + 1))
					return true;
		}
		return false;
	}

	return check(obj, 1);
}
