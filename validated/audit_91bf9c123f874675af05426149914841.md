### Title
SSRF via unvalidated ArbStore URL fetched for an attacker-chosen `arbiter_address` in arbiter contracts - (File: arbiters.js, arbiter_contract.js)

### Summary
When a wallet interacts with an arbitrated contract (getting arbiter info, opening a dispute, appealing, or requesting an appeal fee), it resolves the `arbiter_address` to a URL via the hub (`hub/get_arbstore_url`) and then issues outbound HTTP(S) requests to that URL with no validation of the destination host/scheme, mirroring the Keycloak jwks_uri SSRF pattern where a caller-controlled address is blindly trusted and fetched.

### Finding Description
`arbiter_address` is a field of the arbiter contract that is fully controlled by the remote counterparty when they send an `arbiter_contract_offer` device message; it is inserted into `wallet_arbiter_contracts` without any validation beyond being a syntactically valid address [1](#0-0) . Any code path that later needs the arbiter's service URL takes this attacker-influenced `arbiter_address`, asks the hub for the corresponding URL, and then treats the returned string as a fully trusted destination:

- `arbiters.getInfo()` and `arbiters.getArbstoreInfo()` only check that the URL returned by the hub is a non-empty string, then immediately perform `http.get(url + ...)`: [2](#0-1) [3](#0-2) [4](#0-3) 
- `arbiter_contract.js` reuses the same unvalidated URL to POST full contract/dispute data via `httpRequest()` in `openDispute()`, `appeal()`, and `getAppealFee()`: [5](#0-4) [6](#0-5) [7](#0-6) 
- `httpRequest()` itself performs no scheme/host allow-listing — it simply parses the URL and issues the request: [8](#0-7) 

There is no restriction preventing the resolved URL from pointing to `localhost`, internal LAN addresses, or cloud metadata endpoints (e.g. `169.254.169.254`), nor any enforcement that the scheme/port match an expected public ArbStore API. The only gate is `isNonemptyString(url)`, exactly analogous to Keycloak accepting any `jwks_uri` string without validating its destination.

### Impact Explanation
A malicious counterparty who proposes an arbiter contract naming an `arbiter_address` they control (or for which they can otherwise influence the hub's `hub/get_arbstore_url` / `hub/get_arbstore_address` mapping) can cause the victim's node to issue outbound HTTP(S) requests — including POST requests carrying contract/dispute details such as `my_address`, `peer_address`, amounts, asset, and pairing codes — to an arbitrary attacker-chosen destination once the victim calls `openDispute`, `appeal`, or `getAppealFee`, or simply looks up arbiter info. This enables SSRF-style probing of internal/private network resources or cloud metadata services reachable from the victim's node, and leaks private contract/payment metadata to the attacker-controlled endpoint. This does not itself cause fund loss but is a genuine information-disclosure/reconnaissance risk triggered entirely by an unprivileged private-payment counterparty.

### Likelihood Explanation
Likelihood is moderate: it requires the victim to accept an arbiter contract from a counterparty and later exercise a dispute/appeal-related action, and it depends on the trust model of `hub/get_arbstore_url` (which I could not fully verify server-side, since the hub implementation of this command lives outside the files reachable in this trace) actually returning attacker-influenced data for an attacker-chosen `arbiter_address`. Regardless, the client-side code performs no independent validation of the fetched URL, so the SSRF exposure exists whenever the returned URL is untrusted.

### Recommendation
Validate any URL obtained from the hub before making outbound requests in `arbiters.js` and `arbiter_contract.js`: enforce `https:` scheme, resolve and reject private/loopback/link-local/metadata IP ranges, and restrict the destination host to an allow-list or a well-known ArbStore domain pattern before calling `http.get`/`httpRequest`.

### Proof of Concept
1. Attacker runs (or claims control of) an ArbStore registered under some `arbiter_address`, causing the hub to answer `hub/get_arbstore_url` for that address with `http://169.254.169.254/latest/meta-data/` (or any internal LAN URL).
2. Attacker sends victim an `arbiter_contract_offer` with `arbiter_address` set to that address [9](#0-8) .
3. Victim accepts the contract normally, pays, then later calls `openDispute(hash, cb)` or `appeal(hash, cb)`.
4. The victim's node calls `device.requestFromHub("hub/get_arbstore_url", objContract.arbiter_address, ...)`, receives the attacker's URL, and issues `httpRequest(url, "/api/dispute/new", ...)`/`httpRequest(url, "/api/appeal/new", ...)`, sending an outbound POST to the attacker-chosen internal address with contract details in the body [6](#0-5) [8](#0-7) .

function appeal(hash, cb) {

		if (objContract.status !== "dispute_resolved")
			return cb("contract can't be appealed");
		var command = "hub/get_arbstore_url";

		if (objContract.arbstore_address) {
			command = "hub/get_arbstore_url_by_address";
			address = objContract.arbstore_address;

				return cb("can't get arbstore url:", err);

				return cb(err);

					contract_hash: hash,
					my_pairing_code: objContract.my_pairing_code,
					my_address: objContract.my_address,

function httpRequest(host, path, data, cb) {
	var reqParams = Object.assign(url.parse(host),
		{
			path: path,
			method: "POST",
			headers: {
				"Content-Type": "application/json",
				"Content-Length": (new TextEncoder().encode(data)).length
			}
		}
	);
	var req = http.request(
		reqParams,
		function(resp){
			var data = "";
			resp.on("data", function(chunk){
				data += chunk;
			});
			resp.on("end", function(){
				try {
					data = JSON.parse(data);
					if (data.error) {
						return cb(data.error);
					}
					cb(null, data);
				} catch (e) {
					cb(e);
				}
			});
		}).on("error", cb);
	req.write(data);
	req.end();
}

function getInfo(address, cb) {
	cb = cb || function() {};
	db.query("SELECT device_pub_key, real_name FROM wallet_arbiters WHERE arbiter_address=?", [address], function(rows){
		if (rows.length && rows[0].real_name) { // request again if no real name
			cb(null, rows[0]);
		} else {
			device.requestFromHub("hub/get_arbstore_url", address, function(err, url){
				if (err) {
					return cb(err);
				}
				if (!validationUtils.isNonemptyString(url))
					return cb("invalid url received from hub");
				requestInfoFromArbStore(url+'/api/arbiter/'+address, function(err, info){
					if (err) {
						return cb(err);
					}
					if (!validationUtils.isNonemptyObject(info))
						return cb("invalid arbiter info received from arbstore");
					db.query("REPLACE INTO wallet_arbiters (arbiter_address, device_pub_key, real_name) VALUES (?, ?, ?)", [address, info.device_pub_key, info.real_name], function() {cb(null, info);});
				});
			});
		}
	});
}

function getArbstoreInfo(arbiter_address, cb) {
	if (!cb)
		return new Promise(function(resolve, reject){
			getArbstoreInfo(arbiter_address, function(err, info){
				if (err) return reject(err);
				resolve(info);
			});
		});
	if (arbStoreInfos[arbiter_address]) return cb(null, arbStoreInfos[arbiter_address]);
	device.requestFromHub("hub/get_arbstore_url", arbiter_address, function(err, url){
		if (err) {
			return cb(err);
		}
		if (!validationUtils.isNonemptyString(url))
			return cb("invalid url received from hub");
		requestInfoFromArbStore(url+'/api/get_info', function(err, info){
			if (err)
				return cb(err);
			if (!validationUtils.isNonemptyObject(info))
				return cb("invalid info received from arbstore");
			const cut = parseFloat(info.cut);
			if (!info.address || !validationUtils.isValidAddress(info.address) || isNaN(cut) || cut < 0 || cut >= 1) {
				return cb("malformed info received from ArbStore");
			}
			info.url = url;
			arbStoreInfos[arbiter_address] = info;
			cb(null, info);
		});
	});
}
