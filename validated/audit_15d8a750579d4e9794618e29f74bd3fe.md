### Title
Unhandled NULL/non-object dereference in `extractAddressPathsFromDefinition` crashes wallet on crafted `new_shared_address` device message - (File: wallet_defined_by_addresses.js)

### Summary
CVE-2017-12950 is a NULL-pointer dereference in libgig's `gig::Region::Region` caused by not validating a structure before dereferencing a member field, letting a crafted file crash the parser. The analogous defect in ocore is in `extractAddressPathsFromDefinition()` (`wallet_defined_by_addresses.js:339-375`), which walks an untrusted address-definition array and, for the `'weighted and'` operator, blindly accesses `args.set[i].value` without checking that `args.set[i]` is actually an object. [1](#0-0) 

### Finding Description
`extractAddressPathsFromDefinition` recursively traverses a definition array supplied entirely by a remote correspondent (paired device). For `'or'`, `'and'`, and `'r of set'` it defensively checks `Array.isArray(args)` / `Array.isArray(args.set)` before indexing, but for `'weighted and'` it only checks that `args.set` is an array — it never validates that each element `args.set[i]` is itself a non-null object before reading `.value` from it: [2](#0-1) 

This function is invoked directly from `handleNewSharedAddress(body, callbacks)`, the handler for an incoming `"new_shared_address"` message from a correspondent (paired) device: [3](#0-2) 

At the point `extractAddressPathsFromDefinition(body.definition)` is called (line 396), only two checks have been performed on `body.definition`: that it is an array of length 2 (`isArrayOfLength`) and that its chash matches `body.address`. Nothing yet enforces that inner nodes such as `weighted and`'s `set` entries are well-formed objects. A remote peer can therefore craft a definition such as:
```
["weighted and", {"set": [null, {"value": ["sig", {...}]}]}]
```
whose top-level chash is made to match `body.address` (the attacker fully controls both `definition` and `address` fields of the message, so the chash check is trivially satisfiable — they just compute the chash of their own crafted definition). When `traverse` reaches `args.set[0].value`, `args.set[0]` is `null`, producing an uncaught `TypeError: Cannot read properties of null (reading 'value')`.

Because `handleNewSharedAddress` has no `try/catch` around this call, the exception propagates up out of the message-processing callback chain. This is structurally the same bug class as CVE-2017-12950: a parser/structure-walker dereferences an inner field of an untrusted, attacker-supplied composite structure without first validating that the field is a non-null object.

### Impact Explanation
An uncaught exception thrown while processing an incoming device message can crash the Node.js process (unhandled exception in an async callback bypasses normal error handling and typically terminates the process), producing a remote, unauthenticated-content denial of service against any wallet/hub node that has this correspondent device paired and is willing to process shared-address proposals. This matches the "node unable to process valid requests / crash" impact class allowed by the rules, triggered purely by a paired device sending one malformed message — no private key leakage or privileged access required.

### Likelihood Explanation
Likelihood is high for any node with device-message correspondents: pairing devices is a normal wallet workflow, and the attack requires sending one `new_shared_address` message with a crafted `weighted and` clause. No signature validation or definition-authentifier check happens before `extractAddressPathsFromDefinition` runs, so the malformed structure reaches the vulnerable code before any deeper `validateAddressDefinition` (full oscript-definition grammar validation) call at line 409, which never gets the chance to reject it first.

### Recommendation
In `extractAddressPathsFromDefinition`, before dereferencing `args.set[i].value` for `'weighted and'`, validate that each `args.set[i]` is a non-null object (e.g. `ValidationUtils.isNonemptyObject(args.set[i])`) exactly as is already done implicitly for the other branches, and bail out of the traversal (or skip the element) if not. Additionally, wrap the call in `handleNewSharedAddress` in a try/catch (as already done for `objectHash.getChash160`) so any unexpected structural error is turned into a graceful `callbacks.ifError(...)` instead of an uncaught exception.

### Proof of Concept
1. Attacker device is paired as a correspondent with the victim wallet.
2. Attacker sends a `new_shared_address` message:
```json
{
  "subject": "new_shared_address",
  "body": {
    "address": "<chash160 of the definition below>",
    "definition": ["weighted and", {"set": [null, {"value": ["sig", {"pubkey": "..."}]}]}],
    "signers": { "r": {"address": "SOME_VALID_ADDR"} }
  }
}
```
3. The victim's handler calls `handleNewSharedAddress`, which calls `extractAddressPathsFromDefinition(body.definition)` at line 396.
4. `traverse` reaches the `'weighted and'` case, executes `traverse(args.set[0].value, ...)` where `args.set[0]` is `null`, throwing `TypeError: Cannot read properties of null (reading 'value')`, uncaught, crashing the wallet process. [4](#0-3) [5](#0-4)

function extractAddressPathsFromDefinition(arrDefinition) {
	var result = {};
	function traverse(arr, path) {
		if (!Array.isArray(arr) || arr.length < 2) return;
		var op = arr[0];
		var args = arr[1];
		switch (op) {
			case 'or':
			case 'and':
				if (Array.isArray(args))
					for (var i = 0; i < args.length; i++)
						traverse(args[i], path + '.' + i);
				break;
			case 'r of set':
				if (args && Array.isArray(args.set))
					for (var i = 0; i < args.set.length; i++)
						traverse(args.set[i], path + '.' + i);
				break;
			case 'weighted and':
				if (args && Array.isArray(args.set))
					for (var i = 0; i < args.set.length; i++)
						traverse(args.set[i].value, path + '.' + i);
				break;
			case 'address':
				result[path] = args;
				break;
			case 'hash':
				result[path] = 'secret';
				break;
			case 'in merkle':
				result[path] = ''; // empty address
				break;
		}
	}
	traverse(arrDefinition, 'r');
	return result;
}

function handleNewSharedAddress(body, callbacks){
	if (!ValidationUtils.isArrayOfLength(body.definition, 2))
		return callbacks.ifError("invalid definition");
	if (typeof body.signers !== "object" || Object.keys(body.signers).length === 0)
		return callbacks.ifError("invalid signers");
	try {
		var addr = objectHash.getChash160(body.definition);
	}
	catch (e) {
		return callbacks.ifError("invalid definition: " + e);
	}
	if (body.address !== addr)
		return callbacks.ifError("definition doesn't match its c-hash");
	for (var signing_path in body.signers){
		var signerInfo = body.signers[signing_path];
		if (signerInfo.address && signerInfo.address !== 'secret' && !ValidationUtils.isValidAddress(signerInfo.address))
			return callbacks.ifError("invalid member address: " + JSON.stringify(signerInfo.address));
	}
	const assocDefinitionAddresses = extractAddressPathsFromDefinition(body.definition);
	for (let signing_path in body.signers) {
		const signerInfo = body.signers[signing_path];
		if (assocDefinitionAddresses[signing_path] !== signerInfo.address)
			return callbacks.ifError("signer address at path " + signing_path + " doesn't match definition");
	}
	for (let def_path in assocDefinitionAddresses) {
		if (!body.signers[def_path])
			return callbacks.ifError("no signer for definition address at path " + def_path);
	}
	determineIfIncludesMeAndRewriteDeviceAddress(body.signers, function(err){
		if (err)
			return callbacks.ifError(err);
		validateAddressDefinition(body.definition, function(err){
			if (err)
				return callbacks.ifError(err);
			addNewSharedAddress(body.address, body.definition, body.signers, body.forwarded, callbacks.ifOk);
		});
	});
}
