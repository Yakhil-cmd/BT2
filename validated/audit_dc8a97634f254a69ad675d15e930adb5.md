### Title
Stack exhaustion via unbounded recursion in `getSourceString`/`extractComponents` when validating signed messages that skip depth checks - ([File: string_utils.js], [File: signed_message.js])

### Summary
`string_utils.js`'s `getSourceString()` (used by `object_hash.js` for `getUnitHashToSign`, `getSignedPackageHashToSign`, `getChash160`, etc.) recurses into nested arrays/objects via `extractComponents()` with **no depth limit**, mirroring the `cplus_demangle_type` self-recursion pattern in the CVE: unbounded recursion driven purely by nesting depth of attacker-supplied data.

### Finding Description
`extractComponents()` in `string_utils.js:13-56` recurses once per nesting level of an object/array with no depth cap. The only depth/size guard in the codebase, `isTooDeeplyNestedOrHasTooManyNodes()` (`string_utils.js:260-284`), is invoked from the main unit-validation path in `validation.js:154` before `getUnitHash`/further processing. However, `signed_message.js`'s `validateSignedMessage` flow (`signed_message.js:255-303`) calls `objectHash.getSignedPackageHashToSign(objSignedMessage)` and subsequently `Definition.validateAuthentifiers` directly on `objSignedMessage`, without first calling `isTooDeeplyNestedOrHasTooManyNodes` on it. Since `getSignedPackageHashToSign` → `getSourceString`/`getJsonSourceString` recurse over the entire signed-message object tree, a message with sufficiently deep array/object nesting (e.g., `[[[[...]]]]`) drives the recursion depth to match the nesting depth, exhausting the call stack (`RangeError: Maximum call stack size exceeded`) before any size/complexity check is reached. Any unprivileged party constructing an arbitrary signed message (device message, private-payment counterparty message, or similar `signed_message` structure) that is fed to `validateSignedMessage` can trigger this. ### Impact Explanation
An unhandled stack-overflow exception in a Node.js process is not caught by ordinary `try/catch` in all V8 configurations reliably and can crash the node process handling the signed-message validation, causing a denial of service for that node/wallet — effectively "a network unable to confirm new units" for the affected node while it is repeatedly fed such payloads, since crashing on this path halts unit/message processing for that peer.

### Likelihood Explanation
Medium-High: constructing a deeply nested array/object structure is trivial and small in byte size (e.g., 10,000 levels of `[` costs only ~10KB), and the message is processed before any depth-limiting check comparable to `isTooDeeplyNestedOrHasTooManyNodes` is applied on this particular code path.

### Recommendation
Apply `isTooDeeplyNestedOrHasTooManyNodes` (or an equivalent recursion-depth guard) uniformly to every object that is passed into `getSourceString`/`getJsonSourceString`/`objectHash.*` functions — specifically before `getSignedPackageHashToSign` is called in `signed_message.js`, and ideally inside `getSourceString`/`getJsonSourceString` themselves as a defense-in-depth measure, converting deep recursion into an explicit bounded loop or rejecting inputs beyond a safe depth.

### Proof of Concept
1. Construct a `signed_message` object whose `.signed_message` (or another field object structure) contains deeply nested arrays, e.g. `signed_message: JSON.parse('['.repeat(50000) + '1' + ']'.repeat(50000))`, with valid `authors` metadata so it passes early structural checks.
2. Submit it to `validateSignedMessage` (e.g., via the light-wallet API path that accepts signed messages).
3. `getSignedPackageHashToSign` → `getSourceString`/`getJsonSourceString` recurse ~50,000 times, exceeding V8's default stack limit and throwing/crashing before any size-based rejection occurs.

function getSourceString(obj) {
	var arrComponents = [];
	function extractComponents(variable){
		if (variable === null)
			throw Error("null value in "+JSON.stringify(obj));
		switch (typeof variable){
			case "string":
				if (variable.includes(STRING_JOIN_CHAR))
					throw Error("00 byte in string value in " + JSON.stringify(obj));
				arrComponents.push("s", variable);
				break;
			case "number":
				if (!isFinite(variable))
					throw Error("invalid number: " + variable);
				arrComponents.push("n", variable.toString());
				break;
			case "boolean":
				arrComponents.push("b", variable.toString());
				break;
			case "object":
				if (Array.isArray(variable)){
					if (variable.length === 0)
						throw Error("empty array in "+JSON.stringify(obj));
					arrComponents.push('[');
					for (var i=0; i<variable.length; i++)
						extractComponents(variable[i]);
					arrComponents.push(']');
				}
				else{
					var keys = Object.keys(variable).sort();
					if (keys.length === 0)
						throw Error("empty object in "+JSON.stringify(obj));
					keys.forEach(function(key){
						if (typeof variable[key] === "undefined")
							throw Error("undefined at "+key+" of "+JSON.stringify(obj));
						if (key.includes(STRING_JOIN_CHAR))
							throw Error("00 byte in object key in " + JSON.stringify(obj));
						arrComponents.push(key);
						extractComponents(variable[key]);
					});
				}
				break;
			default:
				throw Error("getSourceString: unknown type="+(typeof variable)+" of "+variable+", object: "+JSON.stringify(obj));
		}
	}

	extractComponents(obj);
	return arrComponents.join(STRING_JOIN_CHAR);
