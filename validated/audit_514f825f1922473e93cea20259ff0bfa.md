CELLS_PER_EXT_BLOB and FIELD_ELEMENTS_PER_EXT_BLOB are fixed compile-time constants (FIELD_ELEMENTS_PER_BLOB=4096, extended to FIELD_ELEMENTS_PER_EXT_BLOB=8192, CELLS_PER_EXT_BLOB=128), all powers of two, defined at build time and not influenced by any attacker-controlled input. `reverse_bits_limited(n, value)` computes `reverse_bits(value) >> (64 - log2_pow2(n))`, which is a bijection on `[0, n)` whenever `n` is a power of two [1](#0-0) [2](#0-1) . Since `CELLS_PER_EXT_BLOB` is a compiled constant and not attacker-supplied, `reverse_bits_limited(CELLS_PER_EXT_BLOB, i)` is injective for `i` in `[0, CELLS_PER_EXT_BLOB)` on every build; there is no "implementation variance" reachable through the public API that changes this constant per call. uint64_t reverse_bits_limited(uint64_t n, uint64_t value) {
    size_t unused_bit_len = 64 - log2_pow2(n);
    return reverse_bits(value) >> unused_bit_len;
}

static C_KZG_RET compute_vanishing_polynomial_from_roots(
    fr_t *poly, size_t *poly_len, const fr_t *roots, size_t roots_len
) {
    fr_t neg_root;

    if (roots_len == 0) {
        return C_KZG_BADARGS;
    }

    /* Initialize with -root[0] */
    fr_neg(&poly[0], &roots[0]);

    for (size_t i = 1; i < roots_len; i++) {
        fr_neg(&neg_root, &roots[i]);

        poly[i] = neg_root;
        fr_add(&poly[i], &poly[i], &poly[i - 1]);

        for (size_t j = i - 1; j > 0; j--) {
            fr_mul(&poly[j], &poly[j], &neg_root);
            fr_add(&poly[j], &poly[j], &poly[j - 1]);
        }
        fr_mul(&poly[0], &poly[0], &neg_root);
    }

    poly[roots_len] = FR_ONE;
    *poly_len = roots_len + 1;

    return C_KZG_OK;
}
