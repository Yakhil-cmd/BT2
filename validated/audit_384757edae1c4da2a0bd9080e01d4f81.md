### Title
Denial of Service via Unbounded Recursive Serialization of Attacker-Controlled Definitions/Payloads - (File: `string_utils.js`, `object_hash.js`)

### Summary
`getSourceString()` in `string_utils.js` recursively walks an arbitrary JS object with no depth limit, similar to Jansson's unbounded recursive JSON parser that caused CVE-2016-4425. This function is invoked by `object_hash.js`'s `getChash160`/`getBase64Hash`/`getHexHash` on attacker-supplied structures (address definitions, AA definitions, asset conditions) that reach validation before any depth-bounding check occurs, allowing a crafted, deeply nested definition to crash any node processing the unit.

### Finding Description
`getSourceString` recursively calls `extractComponents()` on every nested array/object element with no depth or node-count guard: [1](#0-0) 

By contrast, the same file defines depth/node-count bounded helpers, `isTooDeeplyNestedOrHasTooManyNodes` and `isTooBigObj`, which are used elsewhere (formula evaluation, AA composer, network, wallet) to protect against exactly this recursion class: [2](#0-1) 

However, these guards are not applied before `object_hash.js` calls `getSourceString`/`getJsonSourceString` for hashing: [3](#0-2) 

The most directly reachable path is address/author definition validation. `validateAuthor` in `validation.js` computes `objectHash.getChash160(arrAddressDefinition)` on the raw `objAuthor.definition` taken straight from the posted unit, *before* `Definition.validateDefinition` performs any structural/complexity-bounded walk of it: [4](#0-3) 

Because `getChash160` unconditionally serializes the entire object graph via `getSourceString`, an attacker can post a unit whose `author.definition` (or an asset `issue_condition`/`transfer_condition`, or a nested AA `definition` message payload) is a deeply nested array (e.g., `["r",["r",["r", ... ]]]`, thousands of levels deep). `extractComponents` will recurse once per nesting level with no `MAX_DEPTH` check, unlike the complexity-bounded `evaluate()` walkers in `definition.js`/`aa_validation.js` that do enforce `MAX_DEPTH`/`MAX_COMPLEXITY`: [5](#0-4) [6](#0-5) 

Because native JSON parsing in V8 (used to deserialize the incoming unit) tolerates thousands of nesting levels far beyond what the ~1MB default Node.js thread stack can support for a subsequent native recursive JS function call chain, `extractComponents` will exhaust the call stack and crash the process with an uncaught `RangeError: Maximum call stack size exceeded` (or a hard segfault depending on engine/stack settings) before any semantic definition validation or complexity limiting ever executes.

### Impact Explanation
Any node (full node, hub, or light-serving node) that receives and processes a unit with such a definition crashes while computing `getChash160`, before op-count/complexity limits in `Definition.validateDefinition` are applied. Since this occurs during ordinary unit validation triggered by a single posted unit (not a privileged or peer-trust operation), it can be used to repeatedly crash all nodes that attempt to validate/relay the malicious unit, preventing the network from confirming new units — matching the "network unable to confirm new units" impact class.

### Likelihood Explanation
The bug is trivially reachable by any unprivileged unit poster: crafting `author.definition` (or `issue_condition`/`transfer_condition`/nested AA `definition`) as a deeply right-nested 2-element array requires no special privileges, keys, or timing, and the vulnerable `getChash160` call happens unconditionally as part of standard `validateAuthor` processing.

### Recommendation
Add a depth/node-count guard (reusing the existing `isTooDeeplyNestedOrHasTooManyNodes`/`isTooBigObj` helpers in `string_utils.js`) before calling `objectHash.getChash160`/`getBase64Hash`/`getSourceString` on any attacker-supplied structure (`author.definition`, asset conditions, AA definitions, message payloads), rejecting the unit early if the limit is exceeded. Alternatively, convert `extractComponents` to an iterative (explicit-stack) implementation with an enforced maximum depth consistent with `MAX_DEPTH`/`MAX_COMPLEXITY` used elsewhere in `definition.js`/`aa_validation.js`.

### Proof of Concept
1. Construct a unit with a single author whose `address` chash matches a deeply nested definition, e.g. programmatically build `def = ["sig", {pubkey: "..."}]`; then wrap: `def = ["and", [def, ["and", [def, ["and", [ ... ]]]]]]` repeated tens of thousands of times (or simpler self-referential nesting `["not", innerDef]` repeated N times) so `objAuthor.definition` is a JS array nested to depth N (e.g., N = 50,000).
2. Compute `chash = objectHash.getChash160(def)` locally (attacker's own node, with a raised `--stack-size`, or by tuning N to a depth that still succeeds locally but exceeds the victim's default stack) and post a unit where `objAuthor.address` corresponds to this chash and `objAuthor.definition = def`.
3. Send/broadcast the unit; any node calling `validateAuthor` → `objectHash.getChash160(arrAddressDefinition)` (`validation.js:1472`) invokes `getSourceString` (`string_utils.js:11`), which recurses N times with no depth check and crashes the process before `Definition.validateDefinition`'s `MAX_DEPTH`/`MAX_COMPLEXITY` checks (`definition.js`) are ever reached.

function getSourceString(obj) {
	var arrComponents = [];
	function extractComponents(variable){
		if (variable === null)
			throw Error("null value in "+JSON.stringify(obj));
		switch (typeof variable){
			case "string":
				if (variable.includes(STRING_JOIN_CHAR))
					throw Error("00 byte in string value in " + JSON.stringify(obj));
				arrComponents.push("s", variable);
				break;
			case "number":
				if (!isFinite(variable))
					throw Error("invalid number: " + variable);
				arrComponents.push("n", variable.toString());
				break;
			case "boolean":
				arrComponents.push("b", variable.toString());
				break;
			case "object":
				if (Array.isArray(variable)){
					if (variable.length === 0)
						throw Error("empty array in "+JSON.stringify(obj));
					arrComponents.push('[');
					for (var i=0; i<variable.length; i++)
						extractComponents(variable[i]);
					arrComponents.push(']');
				}
				else{
					var keys = Object.keys(variable).sort();
					if (keys.length === 0)
						throw Error("empty object in "+JSON.stringify(obj));
					keys.forEach(function(key){
						if (typeof variable[key] === "undefined")
							throw Error("undefined at "+key+" of "+JSON.stringify(obj));
						if (key.includes(STRING_JOIN_CHAR))
							throw Error("00 byte in object key in " + JSON.stringify(obj));
						arrComponents.push(key);
						extractComponents(variable[key]);
					});
				}
				break;
			default:
				throw Error("getSourceString: unknown type="+(typeof variable)+" of "+variable+", object: "+JSON.stringify(obj));
		}
	}

	extractComponents(obj);
	return arrComponents.join(STRING_JOIN_CHAR);
}

function isTooDeeplyNestedOrHasTooManyNodes(obj, depthLimit = 100, nodesLimit = 10000) {
	let nodeCount = 0;

	function check(variable, depth){
		if (depth > depthLimit || nodeCount > nodesLimit)
			return true;
		if (variable === null || typeof variable !== "object")
			return false;
		if (Array.isArray(variable)) {
			nodeCount += variable.length;
			for (let v of variable)
				if (check(v, depth + 1))
					return true;
		}
		else {
			nodeCount += Object.keys(variable).length;
			for (let key in variable)
				if (check(variable[key], depth + 1))
					return true;
		}
		return false;
	}

	return check(obj, 1);
}

function isTooBigObj(obj, { depthLimit = 100, nodesLimit = 10000, lengthLimit = 1000000 }) {
	let nodeCount = 0;
	let length = 0;

	function check(variable, depth){
		if (depth > depthLimit || nodeCount > nodesLimit || length > lengthLimit)
			return true;
		if (typeof variable === "string")
			length += variable.length;
		else if (typeof variable === "number" || typeof variable === "boolean")
			length += variable.toString().length;
		else if (variable === null)
			length += 4; // "null"
		else if (typeof variable !== "object")
			throw Error("isTooBigObj: unexpected type=" + (typeof variable) + " of " + variable);
		if (length > lengthLimit)
			return true;
		if (typeof variable !== "object" || variable === null)
			return false;
		if (Array.isArray(variable)) {
			nodeCount += variable.length;
			for (let v of variable)
				if (check(v, depth + 1))
					return true;
		}
		else {
			const keys = Object.keys(variable);
			nodeCount += keys.length;
			length += keys.reduce((sum, key) => sum + key.length, 0);
			for (let key in variable)
				if (check(variable[key], depth + 1))
					return true;
		}
		return false;
	}

	return check(obj, 1);
}
