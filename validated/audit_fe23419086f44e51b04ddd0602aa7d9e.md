### Title
Unauthenticated CPU exhaustion: unit hash is computed over the full attacker-controlled joint before nesting/size limits are enforced - ([File: validation.js])

### Summary
`validation.js`'s `validate()` computes the full cryptographic hash of an incoming unit (`objectHash.getUnitHash(objUnit)`) and compares it to the claimed `objUnit.unit` **before** it checks whether the unit is oversized or too deeply nested (`isTooDeeplyNestedOrHasTooManyNodes`) or contains malformed strings (`isObjectWellFormed`). This mirrors the Nuxt island-endpoint bug class: expensive parsing/hashing of fully attacker-controlled input happens before the cheap structural/size guard that would otherwise reject it.

### Finding Description
In `validate()`: [1](#0-0) 

The order of operations is:
1. `isStringOfLength(objUnit.unit, constants.HASH_LENGTH)` — a cheap check on the length of the *claimed* hash string only.
2. `objectHash.getUnitHash(objUnit)` is computed and compared to `objUnit.unit`.
3. Only *after* that hash computation does the code call `isTooDeeplyNestedOrHasTooManyNodes(objUnit)` and `isObjectWellFormed(objJoint)` to reject oversized/deeply-nested/malformed units.

`getUnitHash()` is not cheap for arbitrary attacker-supplied content: [2](#0-1) 

It performs a `lodash.cloneDeep()` of the entire unit (`getNakedUnit`), then recursively walks every nested object/array in `messages`/`payload` (with `Object.keys(...).sort()` at every level) to build a source string via `getSourceString`/`getJsonSourceString`, and finally hashes it with SHA-256: [3](#0-2) [4](#0-3) 

Because this hash computation happens *before* `isTooDeeplyNestedOrHasTooManyNodes`/`isObjectWellFormed`, an unprivileged unit poster can submit a joint whose `unit` field is a garbage/mismatched hash (guaranteeing rejection) but whose `messages`/payload content is deeply nested and/or contains a very large number of keys/array elements. The node still pays the full `cloneDeep` + recursive-sort-and-stringify + SHA-256 cost for the whole structure before the mismatch (or the nesting/size check) is ever reported, exactly analogous to Nuxt's island handler hashing the full request body before validating the URL hash.

Since ocore/Nitro-style Node.js validation runs synchronously on the single event loop, this computation blocks concurrent unit processing while it executes.

### Impact Explanation
A low-rate stream of crafted joints with deeply nested/huge `messages` content and deliberately invalid `unit` hashes can force the node to repeatedly perform expensive deep-clone and recursive-stringify/hash work on every submission before any of it is short-circuited by the depth/size guard. This can stall or degrade validation throughput node-wide, delaying or preventing confirmation of legitimate units while the event loop is occupied — falling under "network unable to confirm new units."

### Likelihood Explanation
Reaching `validate()` requires only submitting a unit/joint object; this is explicitly within reach of an "unprivileged unit poster" per the allowed analog categories, not a peer/network/hub-specific issue — the vulnerable code path is the generic unit-validation entry point used whenever any actor submits a new unit. I could not fully confirm from the available context whether an outer size cap (e.g., `MAX_UNIT_LENGTH` enforcement or the exact thresholds inside `isTooDeeplyNestedOrHasTooManyNodes`) is applied at an earlier stage than `validate()` itself; if no such cap precedes hash computation, the likelihood is high, since the cost scales with the size of the JSON structure the attacker fully controls in `objUnit.messages`.

### Recommendation
Reorder the checks in `validate()` so that `isTooDeeplyNestedOrHasTooManyNodes(objUnit)` and `isObjectWellFormed(objJoint)` (and any size limits) run immediately after the cheap `isStringOfLength(objUnit.unit, ...)` check and strictly before `objectHash.getUnitHash(objUnit)` is invoked, so that oversized/deeply-nested/malformed joints are rejected cheaply without paying for a full deep-clone and recursive hash computation.

### Proof of Concept
Submit a unit/joint for validation where:
- `objUnit.unit` is a syntactically valid but incorrect base64 hash of the correct length (passes the length check but fails the hash match).
- `objUnit.messages` contains a payload with many thousands of nested keys/arrays (maximizing the cost of `lodash.cloneDeep` and the recursive `getSourceString`/`getJsonSourceString` walk in `object_hash.js`/`string_utils.js`) while still passing the top-level shape checks that occur before line 133 in `validation.js` (none do, at that point).
- Repeat submissions at a low rate; each one forces a full deep-clone + recursive stringify + SHA-256 over the oversized structure before being rejected, consuming CPU on the shared event loop before `isTooDeeplyNestedOrHasTooManyNodes` ever gets a chance to short-circuit it.

function getSourceString(obj) {
	var arrComponents = [];
	function extractComponents(variable){
		if (variable === null)
			throw Error("null value in "+JSON.stringify(obj));
		switch (typeof variable){
			case "string":
				if (variable.includes(STRING_JOIN_CHAR))
					throw Error("00 byte in string value in " + JSON.stringify(obj));
				arrComponents.push("s", variable);
				break;
			case "number":
				if (!isFinite(variable))
					throw Error("invalid number: " + variable);
				arrComponents.push("n", variable.toString());
				break;
			case "boolean":
				arrComponents.push("b", variable.toString());
				break;
			case "object":
				if (Array.isArray(variable)){
					if (variable.length === 0)
						throw Error("empty array in "+JSON.stringify(obj));
					arrComponents.push('[');
					for (var i=0; i<variable.length; i++)
						extractComponents(variable[i]);
					arrComponents.push(']');
				}
				else{
					var keys = Object.keys(variable).sort();
					if (keys.length === 0)
						throw Error("empty object in "+JSON.stringify(obj));
					keys.forEach(function(key){
						if (typeof variable[key] === "undefined")
							throw Error("undefined at "+key+" of "+JSON.stringify(obj));
						if (key.includes(STRING_JOIN_CHAR))
							throw Error("00 byte in object key in " + JSON.stringify(obj));
						arrComponents.push(key);
						extractComponents(variable[key]);
					});
				}
				break;
			default:
				throw Error("getSourceString: unknown type="+(typeof variable)+" of "+variable+", object: "+JSON.stringify(obj));
		}
	}

	extractComponents(obj);
	return arrComponents.join(STRING_JOIN_CHAR);
}

function getJsonSourceString(obj, bAllowEmpty) {
	let cache = new WeakMap();  // object to stringified result
	function stringify(variable){
		if (variable === null)
			throw Error("null value in "+JSON.stringify(obj));
		switch (typeof variable){
			case "string":
				return toWellFormedJsonStringify(variable);
			case "number":
				if (!isFinite(variable))
					throw Error("invalid number: " + variable);
			case "boolean":
				return variable.toString();
			case "object":
				// return cached result if already processed
				if (cache.has(variable))
					return cache.get(variable);
				let result;
				if (Array.isArray(variable)){
					if (variable.length === 0 && !bAllowEmpty)
						throw Error("empty array in "+JSON.stringify(obj));
					result = '[' + variable.map(stringify).join(',') + ']';
				}
				else{
					var keys = Object.keys(variable).sort();
					if (keys.length === 0 && !bAllowEmpty)
						throw Error("empty object in "+JSON.stringify(obj));
					result = '{' + keys.map(function(key){ return toWellFormedJsonStringify(key)+':'+stringify(variable[key]) }).join(',') + '}';
				}
				cache.set(variable, result);  // memoize for future references
				return result;
			default:
				throw Error("getJsonSourceString: unknown type="+(typeof variable)+" of "+variable+", object: "+JSON.stringify(obj));
		}
	}

	return stringify(obj);
}
