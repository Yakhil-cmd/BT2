### Title
Unbounded recursion in `getSourceString`/`getUnitHash` on attacker-controlled `authors[].definition` before depth check - ([File: string_utils.js])

### Summary
`validate()` in `validation.js` computes the unit hash via `objectHash.getUnitHash(objUnit)` before it runs the `isTooDeeplyNestedOrHasTooManyNodes(objUnit)` depth/size guard. The hash computation recurses into the full (naked/stripped) unit object — including `authors[i].definition`, which is attacker-controlled and not stripped — using `getSourceString`/`getJsonSourceString`, both of which recurse into nested arrays/objects with no depth limit at all. This mirrors the CVE-2020-28196 bug class: recursive decoding/serialization of an attacker-supplied nested structure lacking any recursion bound, reachable by any unprivileged unit poster.

### Finding Description
In `validation.js`: [1](#0-0) 
`objectHash.getUnitHash(objUnit)` is invoked immediately after the basic unit-length check, and only *after* this call does the code run: [2](#0-1) 
i.e., the depth/size guard `isTooDeeplyNestedOrHasTooManyNodes` is applied too late — after the hash-computing recursive traversal has already happened.

`getUnitHash` → `getNakedUnit`/`getStrippedUnit` → `getBase64Hash` → `getSourceString`/`getJsonSourceString`: [3](#0-2) 
`getNakedUnit` deep-clones the unit and deletes only `messages[i].payload`/`payload_uri`; it does **not** touch `authors[i].definition`, which remains part of the object being hashed.

`getSourceString` recurses into arrays/objects with no depth limit or recursion counter: [4](#0-3) 
Likewise `getJsonSourceString` recurses unbounded (its only protection is a `WeakMap` cache for shared references, not a depth limit): [5](#0-4) 

By contrast, the codebase clearly recognizes this exact hazard elsewhere and defends against it — `isTooDeeplyNestedOrHasTooManyNodes`/`isTooBigObj` enforce a depth limit of 100 and a node-count limit of 10000: [6](#0-5) 
and AA formula/definition evaluators enforce explicit recursion-depth caps (`MAX_DEPTH = 100`, `depth > 100` checks) precisely to avoid "Maximum call stack size exceeded": [7](#0-6) [8](#0-7) [9](#0-8) 

The unit hash path, however, does not go through any of these depth/complexity-bounded evaluators — `getSourceString`/`getJsonSourceString` are plain unbounded recursive functions and are called on the raw untrusted `authors[i].definition` field before the ordering that would otherwise reject it via `isTooDeeplyNestedOrHasTooManyNodes`.

### Impact Explanation
An attacker (any unprivileged unit poster) can submit a unit whose `authors[0].definition` is an extremely deeply nested array (e.g., thousands of levels of `['and', [...]]` or similar nesting). When any full/light node calls `validation.js`'s `validate()` on this joint, `objectHash.getUnitHash(objUnit)` at line 133 will recurse through `getSourceString`/`getJsonSourceString` on the deeply nested definition before the depth guard at line 154 has a chance to reject it, causing "Maximum call stack size exceeded" and crashing the Node.js process handling validation. Since every validating node (including witnesses) processes every posted unit, this is a straightforward network-wide, unauthenticated remote DoS: nodes crash while attempting to validate the crafted unit, and the network becomes unable to confirm new units while nodes repeatedly hit the same crash on catch-up/re-broadcast — matching the "network unable to confirm new units" impact bar.

### Likelihood Explanation
High likelihood: the attack requires only crafting and broadcasting a single unit with an oversized nested `authors[].definition` array — no special privileges, keys, or node compromise are needed, only the ability to post a unit (the same capability any wallet user has). No signature validity is required to trigger the crash, since the hash/order-of-checks issue occurs during the very first stages of `validate()`, well before signature/authentifier verification.

### Recommendation
Reorder the checks in `validate()` in `validation.js` so that `isTooDeeplyNestedOrHasTooManyNodes(objUnit)` (and `isObjectWellFormed`) run **before** `objectHash.getUnitHash(objUnit)` is computed. Additionally, harden `getSourceString` and `getJsonSourceString` in `string_utils.js` to take an explicit depth parameter and throw once a maximum depth (e.g., 100, consistent with `isTooDeeplyNestedOrHasTooManyNodes`) is exceeded, so that any future or indirect caller of these serialization functions is protected regardless of call order.

### Proof of Concept
1. Construct a unit joint where `authors[0].definition` is a deeply nested oscript array, e.g. programmatically build:
   `def = ['sig', {pubkey: '<valid base64 pubkey>'}]`, then wrap it thousands of times, e.g. `def = ['and', [def, ['sig', {...}]]]` repeated ~50,000 times (or any array nested to a depth well beyond 100, matching the pattern used in the repo's own stress tests for other recursive evaluators): [10](#0-9) 
2. Set `objUnit.authors = [{ address: <address matching chash of the deeply nested definition>, definition: def, authentifiers: {...} }]`, fill in required fields (`version`, `parent_units`, `messages`, `last_ball`, etc.) to pass the initial `typeof`/field-presence checks.
3. Submit the joint via the network layer to a full node, or call `validation.js`'s `validate(objJoint, callbacks)` directly.
4. Observe that `objectHash.getUnitHash(objUnit)` at `validation.js:133` recurses through `getSourceString`/`getJsonSourceString` on `authors[0].definition` and crashes the process with `RangeError: Maximum call stack size exceeded`, before `isTooDeeplyNestedOrHasTooManyNodes` at `validation.js:154` is ever reached.

function getSourceString(obj) {
	var arrComponents = [];
	function extractComponents(variable){
		if (variable === null)
			throw Error("null value in "+JSON.stringify(obj));
		switch (typeof variable){
			case "string":
				if (variable.includes(STRING_JOIN_CHAR))
					throw Error("00 byte in string value in " + JSON.stringify(obj));
				arrComponents.push("s", variable);
				break;
			case "number":
				if (!isFinite(variable))
					throw Error("invalid number: " + variable);
				arrComponents.push("n", variable.toString());
				break;
			case "boolean":
				arrComponents.push("b", variable.toString());
				break;
			case "object":
				if (Array.isArray(variable)){
					if (variable.length === 0)
						throw Error("empty array in "+JSON.stringify(obj));
					arrComponents.push('[');
					for (var i=0; i<variable.length; i++)
						extractComponents(variable[i]);
					arrComponents.push(']');
				}
				else{
					var keys = Object.keys(variable).sort();
					if (keys.length === 0)
						throw Error("empty object in "+JSON.stringify(obj));
					keys.forEach(function(key){
						if (typeof variable[key] === "undefined")
							throw Error("undefined at "+key+" of "+JSON.stringify(obj));
						if (key.includes(STRING_JOIN_CHAR))
							throw Error("00 byte in object key in " + JSON.stringify(obj));
						arrComponents.push(key);
						extractComponents(variable[key]);
					});
				}
				break;
			default:
				throw Error("getSourceString: unknown type="+(typeof variable)+" of "+variable+", object: "+JSON.stringify(obj));
		}
	}

	extractComponents(obj);
	return arrComponents.join(STRING_JOIN_CHAR);

function getJsonSourceString(obj, bAllowEmpty) {
	let cache = new WeakMap();  // object to stringified result
	function stringify(variable){
		if (variable === null)
			throw Error("null value in "+JSON.stringify(obj));
		switch (typeof variable){
			case "string":
				return toWellFormedJsonStringify(variable);
			case "number":
				if (!isFinite(variable))
					throw Error("invalid number: " + variable);
			case "boolean":
				return variable.toString();
			case "object":
				// return cached result if already processed
				if (cache.has(variable))
					return cache.get(variable);
				let result;
				if (Array.isArray(variable)){
					if (variable.length === 0 && !bAllowEmpty)
						throw Error("empty array in "+JSON.stringify(obj));
					result = '[' + variable.map(stringify).join(',') + ']';
				}
				else{
					var keys = Object.keys(variable).sort();
					if (keys.length === 0 && !bAllowEmpty)
						throw Error("empty object in "+JSON.stringify(obj));
					result = '{' + keys.map(function(key){ return toWellFormedJsonStringify(key)+':'+stringify(variable[key]) }).join(',') + '}';
				}
				cache.set(variable, result);  // memoize for future references
				return result;
			default:
				throw Error("getJsonSourceString: unknown type="+(typeof variable)+" of "+variable+", object: "+JSON.stringify(obj));
		}
	}

	return stringify(obj);

function isTooDeeplyNestedOrHasTooManyNodes(obj, depthLimit = 100, nodesLimit = 10000) {
	let nodeCount = 0;

	function check(variable, depth){
		if (depth > depthLimit || nodeCount > nodesLimit)
			return true;
		if (variable === null || typeof variable !== "object")
			return false;
		if (Array.isArray(variable)) {
			nodeCount += variable.length;
			for (let v of variable)
				if (check(v, depth + 1))
					return true;
		}
		else {
			nodeCount += Object.keys(variable).length;
			for (let key in variable)
				if (check(variable[key], depth + 1))
					return true;
		}
		return false;
	}

	return check(obj, 1);
}

function isTooBigObj(obj, { depthLimit = 100, nodesLimit = 10000, lengthLimit = 1000000 }) {
	let nodeCount = 0;
	let length = 0;

	function check(variable, depth){
		if (depth > depthLimit || nodeCount > nodesLimit || length > lengthLimit)
			return true;
		if (typeof variable === "string")
			length += variable.length;
		else if (typeof variable === "number" || typeof variable === "boolean")
			length += variable.toString().length;
		else if (variable === null)
			length += 4; // "null"
		else if (typeof variable !== "object")
			throw Error("isTooBigObj: unexpected type=" + (typeof variable) + " of " + variable);
		if (length > lengthLimit)
			return true;
		if (typeof variable !== "object" || variable === null)
			return false;
		if (Array.isArray(variable)) {
			nodeCount += variable.length;
			for (let v of variable)
				if (check(v, depth + 1))
					return true;
		}
		else {
			const keys = Object.keys(variable);
			nodeCount += keys.length;
			length += keys.reduce((sum, key) => sum + key.length, 0);
			for (let key in variable)
				if (check(variable[key], depth + 1))
					return true;
		}
		return false;
	}

	return check(obj, 1);
}
